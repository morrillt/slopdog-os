type SortOrder = 'asc' | 'desc';

export const FunnyIndex: React.FC<FunnyIndexProps> = ({ models, conversations }) => {
  const [sortField, setSortField] = useState<SortField>('avgDuration');
  const [sortOrder, setSortOrder] = useState<SortOrder>('asc');

  const stats = React.useMemo(() => {
    const modelStats: Record<string, { 
      funny: number; 
      notFunny: number;
      totalDuration: number;
      totalTTFT: number;
      totalTPS: number;
      runs: number;
    }> = {};

    conversations.forEach((conv) => {
      Object.entries(conv.responses).forEach(([modelId, response]) => {
        if (!modelStats[modelId]) {
          modelStats[modelId] = { 
            funny: 0, 
            notFunny: 0,
            totalDuration: 0,
            totalTTFT: 0,
            totalTPS: 0,
            runs: 0
          };
        }
        modelStats[modelId].runs += 1;
        modelStats[modelId].totalDuration += response.duration || 0;
        modelStats[modelId].totalTTFT += response.ttft || 0;
        modelStats[modelId].totalTPS += response.tps || 0;

        if (response.rating === 'funny') {
          modelStats[modelId].funny += 1;
        } else if (response.rating === 'not_funny') {
          modelStats[modelId].notFunny += 1;
        }
      });
    });

    return Object.entries(modelStats)
      .map(([modelId, stat]) => {
        const model = models.find((m) => m.id === modelId);
        const ratedTotal = stat.funny + stat.notFunny;
        const funnyScore = ratedTotal > 0 ? (stat.funny / ratedTotal) * 100 : 0;
        return {
          id: modelId,
          name: model?.name || modelId,
          funny: stat.funny,
          notFunny: stat.notFunny,
          funnyScore,
          avgDuration: stat.totalDuration / stat.runs,
          avgTTFT: stat.totalTTFT / stat.runs,
          avgTPS: stat.totalTPS / stat.runs,
          runs: stat.runs,
        };
      })
      .sort((a, b) => {
        let comparison = 0;
        if (sortField === 'name') {
          comparison = a.name.localeCompare(b.name);
        } else {
          comparison = (a[sortField] as number) - (b[sortField] as number);
        }
        return sortOrder === 'asc' ? comparison : -comparison;
      });
  }, [models, conversations, sortField, sortOrder]);

  const handleSort = (field: SortField) => {
    if (sortField === field) {
      setSortOrder(sortOrder === 'asc' ? 'desc' : 'asc');
    } else {
      setSortField(field);
      // Default to desc for stats, asc for name and durations
      if (field === 'avgDuration' || field === 'avgTTFT') {
        setSortOrder('asc');
      } else if (field === 'name') {
        setSortOrder('asc');
      } else {
        setSortOrder('desc');
      }
    }
  };

          {sortField === field ? (
            sortOrder === 'asc' ? <ArrowUp className="w-3 h-3" /> : <ArrowDown className="w-3 h-3" />
          ) : (
            <ArrowUpDown className="w-3 h-3" />
          )}
        </div>

## Problem Statement
Despite specifying a `thinking_budget` or `reasoning.max_tokens` parameter in requests to OpenRouter free models, most models consistently exceed the specified token limit for internal reasoning. This leads to high latency and potentially unnecessary token usage for "neurotic" or "conversational" internal monologues.

## Things Attempted

### 1. Unified `reasoning` Parameter
Followed the official [OpenRouter documentation](https://openrouter.ai/docs/guides/best-practices/reasoning-tokens) to use the new unified parameter:
```json
"reasoning": {
  "max_tokens": 100,
  "exclude": false,
  "enabled": true
}
```
**Result:** Partially successful for some newer models, but frequently ignored by others.

### 2. Provider-Specific Fallbacks
Tried including legacy parameters for specific providers (like NVIDIA and Alibaba):
- `include_reasoning: true`
- `thinking_budget: 100`
**Result:** No significant improvement in budget enforcement.

### 3. Top-Level `max_tokens` Buffer
Ensured that the top-level `max_tokens` is strictly larger than `reasoning.max_tokens` (as required by Anthropic and others):
```javascript
body.max_tokens = Math.max(thinkingBudget + 2048, 4096);
```
**Result:** Improved stability (prevented request failures), but did not curb the model's verbosity.

### 4. System Prompt Injection
Appended an explicit directive to the system prompt:
`[IMPORTANT: Keep your internal reasoning/thinking extremely concise. Aim for under 100 tokens of reasoning.]`

## Problem Statement

Despite specifying a `thinking_budget` or `reasoning.max_tokens` parameter in requests to OpenRouter free models, most models consistently exceed the specified token limit for internal reasoning. This leads to high latency and potentially unnecessary token usage for "neurotic" or "conversational" internal monologues.

## Things Attempted

### 1\. Unified `reasoning` Parameter

Followed the official OpenRouter documentation to use the new unified parameter:

"reasoning": {
  "max_tokens": 100,
  "exclude": false,
  "enabled": true
}

**Result:** Partially successful for some newer models, but frequently ignored by others.

### 2\. Provider-Specific Fallbacks

Tried including legacy parameters for specific providers (like NVIDIA and Alibaba):

* `include_reasoning: true`
* `thinking_budget: 100`  
**Result:** No significant improvement in budget enforcement.

### 3\. Top-Level `max_tokens` Buffer

Ensured that the top-level `max_tokens` is strictly larger than `reasoning.max_tokens` (as required by Anthropic and others):

body.max_tokens = Math.max(thinkingBudget + 2048, 4096);

**Result:** Improved stability (prevented request failures), but did not curb the model's verbosity.

### 4\. System Prompt Injection

Appended an explicit directive to the system prompt:  
`[IMPORTANT: Keep your internal reasoning/thinking extremely concise. Aim for under 100 tokens of reasoning.]`  
**Result:** Models generally followed the instruction to _be_ concise, but their internal token counters still often drifted past 100.

### 5\. Multi-Field Extraction

Updated the stream parser to look for reasoning in every possible delta field sent by OpenRouter:

* `delta.reasoning`
* `delta.reasoning_content`
* `delta.thinking`
* `delta.reasoning_details` (new standardized array)  
**Result:** Improved metrics accuracy (we are catching the tokens), but confirmed that the models are indeed over-running.

## State of Union (as of Dec 22, 2025)

The current implementation is **stable but un-enforced**.

* **Stable:** Requests no longer crash, and streaming is robust across all tested free models.
* **Un-enforced:** Models like `nemotron-3-nano` and `deepseek-r1` frequently generate 400-1000+ reasoning tokens even when told to stop at 100.

### Next Steps

* Consider a hard-stop at the proxy level (manually truncating the stream once the budget is reached).
* Await more consistent provider implementation of the OpenRouter unified `reasoning` spec.
* For now, the UI includes a **visual pulse warning** when a model goes over its allotted budget.
