+function calculateSharpeRatio(dailyPnL) {

/**
 * API endpoint to list available analysis datasets
 * Scans src/public/data/analysis/ for directories containing manifest.json
 */
export async function GET() {
  try {
    const analysisDir = path.join(process.cwd(), "src", "public", "data", "analysis");
    
    if (!fs.existsSync(analysisDir)) {
      return NextResponse.json({ analyses: [] });
    }

    const entries = fs.readdirSync(analysisDir, { withFileTypes: true });
    const analyses = [];

    for (const entry of entries) {
      if (entry.isDirectory() && !entry.name.startsWith("_")) {
        const manifestPath = path.join(analysisDir, entry.name, "manifest.json");
        if (fs.existsSync(manifestPath)) {
          try {
            const manifestContent = fs.readFileSync(manifestPath, "utf-8");
            const manifest = JSON.parse(manifestContent);
            
            // Extract strategy info from manifest
            const strategyInfo = manifest.configs?.[0];
            const strategyName = strategyInfo?.strategyName || 
                               (manifest.pipelinename?.includes("algo1.0") ? "Easy Peasy Original" : null) ||
                               manifest.pipelinename;
            
            analyses.push({
              pipelinename: entry.name,
              displayName: strategyName || entry.name,
              dateRange: manifest.startDay && manifest.endDay 
                ? `${manifest.startDay} to ${manifest.endDay}`
                : null,
              totalDays: manifest.totalDays || null,
              configCount: manifest.configs?.length || 0,
              isMultiStrategy: manifest.configs?.some((c: any) => !!c.strategyId) || false,
            });
          } catch (err) {
            // Skip directories with invalid manifests
            console.warn(`Skipping ${entry.name}: invalid manifest`, err);
          }
        }
      }
    }

    // Sort by pipelinename (most recent first if date-based naming)
    analyses.sort((a, b) => b.pipelinename.localeCompare(a.pipelinename));

    return NextResponse.json({ analyses });
  } catch (error: any) {
    console.error("Error listing analyses:", error);
    return NextResponse.json(
      { error: error.message || "Failed to list analyses" },
      { status: 500 }
    );
  }
}
