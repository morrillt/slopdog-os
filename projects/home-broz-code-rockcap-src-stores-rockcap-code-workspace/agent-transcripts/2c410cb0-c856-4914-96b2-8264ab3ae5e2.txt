/**
 * Extract HH:MM from timestamp for assertions.
 */
function hhmm(timestamp: string): string {
  const m = timestamp.match(/T(\d{2}):(\d{2})/);
  if (m) return `${m[1]}:${m[2]}`;
  const m2 = timestamp.match(/(\d{2}):(\d{2})/);
  if (m2) return `${m2[1]}:${m2[2]}`;
  return "00:00";
}

/**
 * Create a minimal synthetic fixture for deterministic testing.
 * All candles are 1-second apart starting at 09:30:00.
 */
function createSyntheticFixture(
  startPrice: number,
  candleCount: number,
  candleModifier?: (index: number, baseCandle: Candle) => Partial<Candle>
): { date: string; startPrice: number; candles: Candle[] } {
  const candles: Candle[] = [];

  for (let i = 0; i < candleCount; i++) {
    const minute = 30 + Math.floor(i / 60);
    const second = i % 60;
    const time = `2025-01-02T09:${minute.toString().padStart(2, "0")}:${second
      .toString()
      .padStart(2, "0")}-05:00`;

    const baseCandle: Candle = {
      time,
      open: startPrice,
      high: startPrice,
      low: startPrice,
      close: startPrice,
      volume: 100,
    };

    const modifications = candleModifier ? candleModifier(i, baseCandle) : {};
    candles.push({ ...baseCandle, ...modifications });
  }

  return { date: "2025-01-02", startPrice, candles };
}

/**

/**
 * Check if a timestamp is before the cutoff time.
 */
function isBeforeCutoff(timestamp: string | number, cutoffTime: string): boolean {
  if (!timestamp) return true;
  const timeStr = typeof timestamp === "string" ? timestamp : new Date(timestamp).toISOString();
  const timeMatch = timeStr.match(/(\d{2}):(\d{2})/);
  if (!timeMatch) return true;
  const [, hour, minute] = timeMatch;
  const candleTime = `${hour}:${minute}`;
  return candleTime <= cutoffTime;
}

/**
 * Step 1: Compute the Opening Range (high and low) from the first N candles.
 *
 * NOTE: Currently treats orbWindowMinutes as candle count (known issue).
 */
function computeOpeningRange(
  candles: Candle[],
  orbWindowMinutes: number,
  startPrice: number
): OpeningRange {
  const windowCandles = Math.max(1, Math.floor(orbWindowMinutes));
  const orbCandles = candles.slice(0, Math.min(windowCandles, candles.length));

  if (orbCandles.length === 0) {
    return { high: startPrice, low: startPrice, candleCount: 0 };
  }

  return {
    high: Math.max(...orbCandles.map(c => c.high)),
    low: Math.min(...orbCandles.map(c => c.low)),
    candleCount: orbCandles.length,
  };
}

/**
 * Step 2: Find a breakout beyond the ORB + buffer.
 *
 * Scans candles starting from scanStartIndex for price breaking above
 * orbHigh + buffer (long) or below orbLow - buffer (short).
 */
function findBreakout(
  state: TradingState,
  scanStartIndex: number
): BreakoutResult | null {
  const { candles, orb, atrParams, config, cutoffTime, log } = state;

  for (let i = scanStartIndex; i < candles.length; i++) {
    const candle = candles[i];

    // Check cutoff
    if (!isBeforeCutoff(candle.time, cutoffTime)) {
      log(`  Candle[${i}] is after cutoff time ${cutoffTime}, stopping`);
      break;
    }

    // Calculate ATR buffer
    const currentAtr = i >= atrParams.period
      ? getAtrAtCandleIndex(candles, atrParams.period, i)
      : atrParams.period; // Fallback if not enough candles
    const breakoutBuffer = currentAtr * config.breakoutBufferAtr;

    const longBreakoutLevel = orb.high + breakoutBuffer;
    const shortBreakoutLevel = orb.low - breakoutBuffer;

    const longHit = candle.high >= longBreakoutLevel;
    const shortHit = candle.low <= shortBreakoutLevel;

    if (longHit || shortHit) {
      // Tiebreaker: choose direction closer to open
      let direction: "long" | "short";
      if (longHit && shortHit) {
        const distToLong = Math.abs(candle.open - longBreakoutLevel);
        const distToShort = Math.abs(candle.open - shortBreakoutLevel);
        direction = distToLong <= distToShort ? "long" : "short";
        log(`  Candle[${i}] BOTH breakouts hit! Tiebreaker -> ${direction}`);
      } else {
        direction = longHit ? "long" : "short";
        log(`  Candle[${i}] ${direction.toUpperCase()} breakout hit`);
      }

      return { candleIndex: i, direction, confirmed: false };
    }
  }

  return null;
}

/**
 * Step 3: Confirm breakout based on confirmationMode.
 *
 * For "close" mode: breakout is already confirmed by high/low touch.
 * For "two_closes" mode: requires next candle to also close beyond breakout level.
 *
 * Returns true if confirmed, false if confirmation failed.
 * IMPORTANT: Does not mutate direction state on failure to prevent leak (AC-8).
 */
function confirmBreakout(
  state: TradingState,
  breakout: BreakoutResult
): boolean {
  const { candles, orb, atrParams, config, log } = state;

  if (config.confirmationMode === "close") {
    // "close" mode doesn't require additional confirmation
    return true;
  }

  // "two_closes" mode: check if next candle closes beyond breakout level
  const nextIndex = breakout.candleIndex + 1;
  if (nextIndex >= candles.length) {
    log(`  Not enough candles for two_closes confirmation, skipping`);
    return false;
  }

  const nextCandle = candles[nextIndex];
  const currentAtr = breakout.candleIndex >= atrParams.period
    ? getAtrAtCandleIndex(candles, atrParams.period, breakout.candleIndex)
    : atrParams.period;
  const breakoutBuffer = currentAtr * config.breakoutBufferAtr;

  if (breakout.direction === "long") {
    const confirms = nextCandle.close >= (orb.high + breakoutBuffer);
    if (!confirms) {
      log(`  Second close failed to confirm LONG breakout, skipping`);
      return false;
    }
  } else {
    const confirms = nextCandle.close <= (orb.low - breakoutBuffer);
    if (!confirms) {
      log(`  Second close failed to confirm SHORT breakout, skipping`);
      return false;
    }
  }

  log(`  Two-closes confirmation succeeded`);
  return true;
}

/**
 * Step 4: Optionally wait for retest of the ORB level.
 *
 * If requireRetest is true, scans for price to return to the ORB boundary
 * within the timeout period.
 *
 * Returns the retest candle index, or -1 if retest not found/not required.
 */
function maybeWaitForRetest(
  state: TradingState,
  breakout: BreakoutResult
): number {
  const { candles, orb, config, log } = state;

  if (!config.requireRetest) {
    return breakout.candleIndex; // No retest required, return breakout index
  }

  log(`  Waiting for retest of ORB level...`);
  const retestLevel = breakout.direction === "long" ? orb.high : orb.low;
  const timeoutCandles = config.retestTimeoutMinutes; // Currently treated as candle count
  const searchEndIndex = Math.min(breakout.candleIndex + timeoutCandles, candles.length - 1);

  for (let i = breakout.candleIndex; i <= searchEndIndex; i++) {
    const candle = candles[i];
    const hasRetest = breakout.direction === "long"
      ? candle.low <= retestLevel  // Price came back down to test the high
      : candle.high >= retestLevel; // Price came back up to test the low

    if (hasRetest) {
      log(`  Retest confirmed at candle[${i}]`);
      return i;
    }
  }

  log(`  No retest found within timeout period. Skipping this signal.`);
  return -1;
}

/**
 * Step 5: Find entry signal (trigger crossing) after breakout/retest confirmation.
 *
 * Scans for price crossing the fixed trigger levels based on startPrice.
 * The direction from breakout gates which trigger we look for.
 */
function findEntrySignal(
  state: TradingState,
  scanStartIndex: number,
  breakoutDirection: "long" | "short"
): EntrySignal | null {
  const { candles, longTrigger, shortTrigger, cutoffTime, log } = state;

  for (let i = scanStartIndex; i < candles.length; i++) {
    const candle = candles[i];

    if (!isBeforeCutoff(candle.time, cutoffTime)) {
      log(`  Candle[${i}] is after cutoff time ${cutoffTime}, stopping`);
      break;
    }

    const longHit = candle.high >= longTrigger;
    const shortHit = candle.low <= shortTrigger;

    if (longHit || shortHit) {
      // Determine actual direction from what was hit
      let direction: "long" | "short";
      if (longHit && shortHit) {
        // Tiebreaker
        const distToLong = Math.abs(candle.open - longTrigger);
        const distToShort = Math.abs(candle.open - shortTrigger);
        direction = distToLong <= distToShort ? "long" : "short";
        log(`  Candle[${i}] BOTH triggers hit! Tiebreaker -> ${direction}`);
      } else {
        direction = longHit ? "long" : "short";
        log(`  Candle[${i}] ${direction.toUpperCase()} trigger hit`);
      }

      const slippage = calculateEntrySlippage(direction);
      const triggerPrice = direction === "long" ? longTrigger : shortTrigger;
      const entryPrice = direction === "long"
        ? triggerPrice + slippage
        : triggerPrice - slippage;

      log(`  Entry: ${entryPrice} at ${candle.time}`);

      return {
        candleIndex: i,
        direction,
        entryPrice,
        entryTimestamp: candle.time,
        slippage,
      };
    }
  }

  return null;
}

/**
 * Step 6: Compute entry price, TP, and SL.
 *
 * Returns TP and SL prices based on direction and fixed offsets from startPrice/trigger.
 */
function computeTpSl(
  state: TradingState,
  direction: "long" | "short"
): { tpPrice: number; slPrice: number } {
  const { startPrice, profitTargetPts, stopLossPts, longTrigger, shortTrigger, log } = state;

  let tpPrice: number;
  let slPrice: number;

  if (direction === "long") {
    tpPrice = startPrice + profitTargetPts;
    slPrice = longTrigger - stopLossPts;
  } else {
    tpPrice = startPrice - profitTargetPts;
    slPrice = shortTrigger + stopLossPts;
  }

  log(`  TP: ${tpPrice}, Initial SL: ${slPrice}`);
  return { tpPrice, slPrice };
}

/**
 * Step 7: Scan for exit condition (TP hit, SL hit, or EOD).
 *
 * Includes intrabar resolution for same-candle exits.
 */
function scanExit(
  state: TradingState,
  entry: EntrySignal,
  tpPrice: number,
  slPrice: number
): ExitResult {
  const { candles, longTrigger, shortTrigger, log } = state;
  const { candleIndex: triggerCandleIndex, direction, entryPrice, entryTimestamp } = entry;

  let exitCandleIndex = -1;
  let exitNote: "PT Hit" | "SL Hit" | "EOD Push" = "EOD Push";
  let exitPrice = entryPrice;
  let exitTimestamp = candles[candles.length - 1].time;
  let exitSlippage = 0;

  for (let i = triggerCandleIndex; i < candles.length; i++) {
    const candle = candles[i];

    // Same-candle intrabar resolution
    if (i === triggerCandleIndex) {
      const entryLevel = direction === "long" ? longTrigger : shortTrigger;
      const intrabarResult = resolveIntrabarDirectional({
        candle: { O: candle.open, H: candle.high, L: candle.low, C: candle.close, V: candle.volume || 0 },
        side: direction,
        entryLevel,
        stopLevel: slPrice,
        targetLevel: tpPrice,
      });

      if (intrabarResult.event === "FILL_STOP_SAME_BAR") {
        exitCandleIndex = i;
        exitTimestamp = candle.time;
        exitNote = "SL Hit";
        exitPrice = intrabarResult.exitFill!;
        log(`  Exit: ${exitNote} at candle[${i}], price=${exitPrice} (intrabar)`);
        break;
      } else if (intrabarResult.event === "FILL_TARGET_SAME_BAR") {
        exitCandleIndex = i;
        exitTimestamp = candle.time;
        exitNote = "PT Hit";
        exitPrice = intrabarResult.exitFill!;
        log(`  Exit: ${exitNote} at candle[${i}], price=${exitPrice} (intrabar)`);
        break;
      }
      continue;
    }

    // Standard exit check for subsequent candles
    let tpHit = false;
    let slHit = false;

    if (direction === "long") {
      tpHit = candle.high >= tpPrice;
      slHit = candle.low <= slPrice;
    } else {
      tpHit = candle.low <= tpPrice;
      slHit = candle.high >= slPrice;
    }

    if (tpHit || slHit) {
      exitCandleIndex = i;
      exitTimestamp = candle.time;

      if (tpHit && slHit) {
        // Conservative: SL wins
        exitNote = "SL Hit";
        exitSlippage = calculateSlSlippage(direction);
        exitPrice = direction === "long" ? slPrice - exitSlippage : slPrice + exitSlippage;
      } else if (slHit) {
        exitNote = "SL Hit";
        exitSlippage = calculateSlSlippage(direction);
        exitPrice = direction === "long" ? slPrice - exitSlippage : slPrice + exitSlippage;
      } else {
        exitNote = "PT Hit";
        exitSlippage = calculateTpSlippage(direction);
        exitPrice = direction === "long" ? tpPrice - exitSlippage : tpPrice + exitSlippage;
      }
      log(`  Exit: ${exitNote} at candle[${i}], price=${exitPrice}`);
      break;
    }
  }

  // Calculate net P&L
  const net = direction === "long"
    ? exitPrice - entryPrice
    : entryPrice - exitPrice;

  if (exitCandleIndex === -1) {
    log(`  Exit: EOD Push at entry price ${exitPrice} (net=0)`);
  }

  return {
    candleIndex: exitCandleIndex,
    price: exitPrice,
    timestamp: exitTimestamp,
    note: exitNote,
    slippage: exitSlippage,
    net,
  };
}

/**
 * Step 8: Find re-eligibility index (price returns between triggers).
 *
 * After a trade exits, price must return to the zone between triggers
 * before another trade can be triggered.
 */
function findReEligibilityIndex(
  state: TradingState,
  exitCandleIndex: number
): number {
  const { candles, longTrigger, shortTrigger, log } = state;

  if (exitCandleIndex === -1) {
    // EOD Push - no more candles
    return -1;
  }

  for (let i = exitCandleIndex + 1; i < candles.length; i++) {
    const candle = candles[i];
    const lowInZone = candle.low > shortTrigger && candle.low < longTrigger;
    const highInZone = candle.high > shortTrigger && candle.high < longTrigger;
    const openInZone = candle.open > shortTrigger && candle.open < longTrigger;
    const closeInZone = candle.close > shortTrigger && candle.close < longTrigger;

    if (lowInZone || highInZone || openInZone || closeInZone) {
      log(`  Re-eligibility found at candle[${i}] - price returned between triggers`);
      return i;
    }
  }

  log(`  No re-eligibility found - price never returned between triggers. Done.`);
  return -1;
}

/**
 * Execute the multi-round-trip trading loop.
 *
 * Orchestrates the helper functions in sequence for each trade attempt.
 */
function executeTradingLoop(
  state: TradingState,
  initialScanStart: number,
  maxRoundTrips: number
): AlgoTradeRow[] {
  const { log } = state;
  const allRows: AlgoTradeRow[] = [];
  let scanStartIndex = initialScanStart;
  let tradeNumber = 0;

  while (tradeNumber < maxRoundTrips && scanStartIndex < state.candles.length) {
    tradeNumber++;
    log(`-`.repeat(70));
    log(`ROUND TRIP #${tradeNumber} - Scanning from candle[${scanStartIndex}]`);

    // Step 2: Find breakout
    const breakout = findBreakout(state, scanStartIndex);
    if (!breakout) {
      log(`  No breakout found from candle[${scanStartIndex}] onwards. Done.`);
      break;
    }

    // Step 3: Confirm breakout
    const confirmed = confirmBreakout(state, breakout);
    if (!confirmed) {
      // Failed confirmation - continue scanning from next candle
      // IMPORTANT: Do not carry over breakout direction (AC-8)
      scanStartIndex = breakout.candleIndex + 1;
      tradeNumber--; // Don't count failed confirmation as a trade attempt
      continue;
    }

    // Step 4: Wait for retest if required
    const retestIndex = maybeWaitForRetest(state, breakout);
    if (retestIndex === -1) {
      // Retest timeout - continue scanning
      scanStartIndex = breakout.candleIndex + 1;
      tradeNumber--;
      continue;
    }

    // Step 5: Find entry signal
    const entryScanStart = state.config.requireRetest ? retestIndex + 1 : breakout.candleIndex + 1;
    const entry = findEntrySignal(state, entryScanStart, breakout.direction);
    if (!entry) {
      log(`  No trigger crossed after breakout confirmation. Done.`);
      break;
    }

    // Step 6: Calculate TP and SL
    const { tpPrice, slPrice } = computeTpSl(state, entry.direction);

    // Step 7: Scan for exit
    const exit = scanExit(state, entry, tpPrice, slPrice);
    log(`  Net P&L: ${exit.net}`);

    // Build trade rows
    const openRow: AlgoTradeRow = {
      id: `open-${tradeNumber}`,
      timestamp: entry.entryTimestamp,
      type: entry.direction === "long" ? "Open Long" : "Open Short",
      price: entry.entryPrice,
      slippage: entry.slippage,
    };

    const closeRow: AlgoTradeRow = {
      id: `close-${tradeNumber}`,
      timestamp: exit.timestamp,
      type: entry.direction === "long" ? "Close Long" : "Close Short",
      price: exit.price,
      slippage: exit.slippage,
      note: exit.note,
      net: exit.net,
    };

    allRows.push(openRow, closeRow);

    // Check for EOD
    if (exit.note === "EOD Push") {
      log(`  EOD reached, no more trades possible`);
      break;
    }

    // Step 8: Find re-eligibility
    const reEligibleIndex = findReEligibilityIndex(state, exit.candleIndex);
    if (reEligibleIndex === -1) {
      break;
    }

    scanStartIndex = reEligibleIndex + 1;
    log(`  Next scan will start from candle[${scanStartIndex}]`);
  }

  return allRows;
}

 * =============================================================================
 * STEPPED STOP LOSS HELPER
 * =============================================================================
 *
 * Shared logic for stepped stop loss (profit-locking) used by multiple strategies.
 * Single source of truth for AC-6 compliance.
 *
 * ## How Stepped SL Works
 *
 * When price moves in your favor, the stop loss "steps up" to lock in profits:
 * - Config: `{ "20": 2, "25": 10 }` means:
 *   - At 20pts profit → lock in 2pts (SL moves to entry + 2)
 *   - At 25pts profit → lock in 10pts (SL moves to entry + 10)
 * - SL only tightens; it never loosens back.
 *
 * @module steppedStopLoss
 */

import type { Candle } from "../../backtest";

// =============================================================================
// TYPES
// =============================================================================

/** Configuration for stepped stop loss */
export interface SteppedSlConfig {
  /** Maps profit threshold (string) to profit to lock (number) */
  thresholds: Record<string, number>;
}

/** State tracking for stepped SL during a trade */
export interface SteppedSlState {
  /** Current SL price (updated as thresholds are crossed) */
  currentSlPrice: number;
  /** Maximum unrealized profit reached (for longs: max high - entry) */
  maxUnrealizedProfit: number;
  /** History of SL changes for UI visualization */
  slChanges: Array<{
    timestamp: string;
    slPrice: number;
    profitLocked: number;
    threshold: number;
  }>;
}

/** Result of processing a candle for stepped SL */
export interface SteppedSlUpdate {
  /** Whether SL was updated on this candle */
  updated: boolean;
  /** New SL price (same as old if not updated) */
  newSlPrice: number;
  /** Current unrealized profit for this candle */
  unrealizedProfit: number;
  /** Profit locked if updated, undefined otherwise */
  profitLocked?: number;
  /** Threshold that was crossed if updated */
  thresholdCrossed?: number;
}

// =============================================================================
// MAIN FUNCTIONS
// =============================================================================

/**
 * Parse and validate stepped SL configuration.
 *
 * @param config - Raw config object like `{"20": 2, "25": 10}`
 * @returns Parsed config with sorted thresholds, or null if no valid config
 */
export function parseSteppedSlConfig(
  config?: Record<string, number>
): { thresholds: number[]; config: Record<string, number> } | null {
  if (!config || typeof config !== "object") {
    return null;
  }

  const thresholds = Object.keys(config)
    .map(k => Number(k))
    .filter(k => !isNaN(k) && k > 0)
    .sort((a, b) => a - b);

  if (thresholds.length === 0) {
    return null;
  }

  return { thresholds, config };
}

/**
 * Initialize stepped SL state for a new trade.
 *
 * @param initialSlPrice - Initial stop loss price
 * @returns Initial state object
 */
export function initSteppedSlState(initialSlPrice: number): SteppedSlState {
  return {
    currentSlPrice: initialSlPrice,
    maxUnrealizedProfit: 0,
    slChanges: [],
  };
}

/**
 * Process a candle for stepped SL updates.
 *
 * Call this for each candle during exit scanning to update SL based on profit.
 *
 * @param candle - Current candle
 * @param entryPrice - Trade entry price
 * @param direction - Trade direction ("long" or "short")
 * @param state - Current stepped SL state (will NOT be mutated)
 * @param config - Stepped SL config (thresholds and profit-to-lock mapping)
 * @returns Update result with new state values
 */
export function processSteppedSl(
  candle: Candle,
  entryPrice: number,
  direction: "long" | "short",
  state: SteppedSlState,
  config: { thresholds: number[]; config: Record<string, number> }
): SteppedSlUpdate {
  // Calculate unrealized profit for this candle
  let unrealizedProfit: number;
  if (direction === "long") {
    // For longs: profit = highest price reached - entry price
    const candleHighProfit = candle.high - entryPrice;
    unrealizedProfit = Math.max(state.maxUnrealizedProfit, candleHighProfit);
  } else {
    // For shorts: profit = entry price - lowest price reached
    const candleLowProfit = entryPrice - candle.low;
    unrealizedProfit = Math.max(state.maxUnrealizedProfit, candleLowProfit);
  }

  let currentSlPrice = state.currentSlPrice;
  let updated = false;
  let profitLocked: number | undefined;
  let thresholdCrossed: number | undefined;

  // Check each threshold (sorted ascending)
  for (const threshold of config.thresholds) {
    if (unrealizedProfit >= threshold) {
      const profitToLock = config.config[String(threshold)];
      if (profitToLock != null) {
        // Calculate new SL price
        const newSlPrice = direction === "long"
          ? entryPrice + profitToLock
          : entryPrice - profitToLock;

        // Only update if new SL is better (higher for longs, lower for shorts)
        const shouldUpdate = direction === "long"
          ? newSlPrice > currentSlPrice
          : newSlPrice < currentSlPrice;

        if (shouldUpdate) {
          currentSlPrice = newSlPrice;
          updated = true;
          profitLocked = profitToLock;
          thresholdCrossed = threshold;
        }
      }
    }
  }

  return {
    updated,
    newSlPrice: currentSlPrice,
    unrealizedProfit,
    profitLocked,
    thresholdCrossed,
  };
}

/**
 * Apply a stepped SL update to state.
 *
 * @param state - Current state (will NOT be mutated)
 * @param update - Update result from processSteppedSl
 * @param timestamp - Candle timestamp for change tracking
 * @returns New state with updates applied
 */
export function applySteppedSlUpdate(
  state: SteppedSlState,
  update: SteppedSlUpdate,
  timestamp: string
): SteppedSlState {
  const newState: SteppedSlState = {
    currentSlPrice: update.newSlPrice,
    maxUnrealizedProfit: update.unrealizedProfit,
    slChanges: [...state.slChanges],
  };

  if (update.updated && update.profitLocked !== undefined && update.thresholdCrossed !== undefined) {
    newState.slChanges.push({
      timestamp,
      slPrice: update.newSlPrice,
      profitLocked: update.profitLocked,
      threshold: update.thresholdCrossed,
    });
  }

  return newState;
}

### Strategy Flow

1. **Define Opening Range**: Calculate high/low from first N minutes
2. **Wait for Breakout**: Price breaks beyond range ± ATR buffer
3. **Confirm Breakout**: Optional extra confirmation depending on `confirmationMode`
   - `confirmationMode: "close"` → breakout is detected when `high/low` crosses the buffered level (no close requirement)
   - `confirmationMode: "two_closes"` → requires the *next candle* to also close beyond the buffered level
4. **Optional Retest**: *After* a breakout is detected, optionally wait for price to **pull back and "touch" the ORB boundary** before allowing an entry.  
   - **Why**: many false breakouts spike through the level and immediately reverse; requiring a retest can filter some of that chop.  
   - **What "retest" means in our current code**: it looks for a candle that touches the **raw ORB level** (not the ATR-buffered breakout level):  
     - Long breakout → retest is `candle.low <= orbHigh`  
     - Short breakout → retest is `candle.high >= orbLow`  
   - **Timeout**: if no retest happens within `retestTimeoutMinutes` (currently treated as a candle-count window), the signal is skipped.
5. **Enter Trade**: Enter when trigger level is crossed after confirmation
6. **Manage Exit**: Use fixed profit target and stop loss, or let trade run to cutoff time

## New Features (TICKET-006)

### Entry Models (`entryModel`)

The strategy supports two entry models that determine how trigger and TP levels are calculated:

- **`"startPriceTriggers"`** (default/baseline):
  - Long trigger = `startPrice + triggerPts`
  - Short trigger = `startPrice - triggerPts`
  - TP = `startPrice ± profitTargetPts`

- **`"orbAnchored"`**:
  - Long trigger = `orbHigh + triggerPts`
  - Short trigger = `orbLow - triggerPts`
  - TP = `orbHigh/orbLow ± profitTargetPts`

### Stepped Stop Loss (`stopLossByProfitThreshold`)

When configured, the stop loss "steps up" to lock in profits as price moves favorably:

```javascript
stopLossByProfitThreshold: { "20": 2, "25": 10 }
// At 20pts profit → SL moves to entry + 2pts (locking 2pts)
// At 25pts profit → SL moves to entry + 10pts (locking 10pts)
```

Key behaviors:
- SL only tightens; it never loosens back
- Profit is measured from entry price
- Uses shared helper: `src/lib/backtest/helpers/steppedStopLoss.ts`

### Config Field Naming (Seconds)

Time-based fields now use `...Seconds` naming (with backward compatibility for `...Minutes`):

| New Name | Legacy Name | Default |
|----------|-------------|---------|
| `orbWindowSeconds` | `orbWindowMinutes` | 5 |
| `retestTimeoutSeconds` | `retestTimeoutMinutes` | 5 |
| `avoidFirstSeconds` | `avoidFirstMinutes` | 0 |

With 1-second candles, seconds equals candle count.

/**
 * Parse comma-separated string values from an env var.
 * Returns array of trimmed non-empty strings.
 *
 * @param {string} envVar - The raw env var value
 * @param {string} name - Name for error messages
 * @returns {string[]} Array of parsed strings
 */
function parseStringList(envVar, name) {
  const values = envVar
    .split(",")
    .map(v => v.trim())
    .filter(v => v.length > 0);

  if (values.length === 0) {
    throw new Error(`${name}: must contain at least one value`);
  }

  return values;
}

/**
 * Parse comma-separated boolean values from an env var.
 * Accepts: "true", "false" (case-insensitive).
 *
 * @param {string} envVar - The raw env var value
 * @param {string} name - Name for error messages
 * @returns {boolean[]} Array of parsed booleans
 */
function parseBooleanList(envVar, name) {
  const values = envVar
    .split(",")
    .map(v => v.trim().toLowerCase())
    .filter(v => v.length > 0)
    .map(v => {
      if (v === "true" || v === "1" || v === "yes") return true;
      if (v === "false" || v === "0" || v === "no") return false;
      throw new Error(`${name}: invalid boolean value "${v}" (use true/false)`);
    });

  if (values.length === 0) {
    throw new Error(`${name}: must contain at least one value`);
  }

  return values;
}

import csv
import json
from collections import defaultdict

rows = []
with open('data/pipeline/4yearpipe1s/stage2 - analysisResults/permutationresults-22.csv', 'r') as f:
    lines = f.readlines()
    start_idx = 0
    for i, line in enumerate(lines):
        if line.startswith('start_day,end_day'):
            start_idx = i
            break
    reader = csv.DictReader(lines[start_idx:])
    for row in reader:
        rows.append(row)

# Group by key parameters
by_orb_window = defaultdict(list)
by_buffer = defaultdict(list)
by_confirm = defaultdict(list)
by_retest = defaultdict(list)
by_entry = defaultdict(list)
by_tp = defaultdict(list)

for r in rows:
    params = json.loads(r['strategy_params'])
    profit = float(r['net_profit_total'])
    
    by_orb_window[params.get('orbWindowSeconds')].append(profit)
    by_buffer[params.get('breakoutBufferAtr')].append(profit)
    by_confirm[params.get('confirmationMode')].append(profit)
    by_retest[params.get('requireRetest')].append(profit)
    by_entry[params.get('entryModel')].append(profit)
    by_tp[r['tp_level']].append(profit)

def avg_profit(group):
    return sum(group) / len(group) if group else 0

print("=== PARAMETER SENSITIVITY ANALYSIS ===\n")

print("ORB Window Seconds:")
for window in sorted(by_orb_window.keys()):
    profits = by_orb_window[window]
    print(f"  {window:>3}s: Avg={avg_profit(profits):>8.2f} | Count={len(profits):>5} | Range=[{min(profits):>8.2f}, {max(profits):>8.2f}]")

print("\nBreakout Buffer ATR:")
for buffer in sorted(by_buffer.keys()):
    profits = by_buffer[buffer]
    print(f"  {buffer:>4}: Avg={avg_profit(profits):>8.2f} | Count={len(profits):>5} | Range=[{min(profits):>8.2f}, {max(profits):>8.2f}]")
