/**
 * Execute the main trading loop that processes multiple round-trip trades.
 * 
 * **What this function does:**
 * This is the core loop that processes multiple round-trip trades. It continues
 * until we run out of candles, hit max round trips, or can't find more triggers.
 * 
 * **Loop Flow:**
 * 1. Check for 9:35 reset (if we've crossed 9:35 without a trade active)
 * 2. Check if we've hit max round trips for current session
 * 3. Scan for trigger crossing (entry signal)
 * 4. If trigger found: calculate entry, scan for exit, build trade rows
 * 5. Scan for re-eligibility (price must return between triggers)
 * 6. Repeat until no more triggers or max round trips reached
 * 
 * **State Management:**
 * This function modifies the state object passed in. In TypeScript/JavaScript,
 * objects are passed by reference, so modifications to the state object are
 * visible to the caller. This is a common pattern for managing complex state.
 * 
 * @param state - Trading state object (will be modified)
 * @param logMain - Main function logger
 * @returns Array of all generated trade rows
 */
function executeTradingLoop(
  state: TradingState,
  logMain: ReturnType<typeof logger.main>
): AlgoTradeRow[] {
  // Main trading loop - continues until we run out of candles or hit limits
  while (state.scanStartIndex < state.candles.length) {
    
    // Check for 9:35 reset logic (before checking limits)
    // If we passed 9:35 while waiting for next trade, reset session count
    check935Reset(state);
    
    // Check if we've hit max round trips for current session
    // If so, try to fast-forward to 9:35 (if not already there) or break
    if (state.sessionTradeCount >= state.maxRoundTrips) {
      if (!state.hasResetAt935 && state.candle935Index >= 0 && state.scanStartIndex < state.candle935Index) {
        state.log(`  Max round trips (${state.maxRoundTrips}) reached for pre-9:35. Fast-forwarding to 9:35.`);
        state.scanStartIndex = state.candle935Index;
        continue; // Restart loop, triggering reset logic above
      }
      state.log(`  Max round trips (${state.maxRoundTrips}) reached for current session. Done.`);
      break;
    }

    // Increment trade counters
    state.tradeNumber++;
    state.sessionTradeCount++;
    state.log("-".repeat(70));
    state.log(`ROUND TRIP #${state.tradeNumber} (Session #${state.sessionTradeCount}) - Scanning from candle[${state.scanStartIndex}]`);
    
    // Scan for trigger crossing (entry signal)
    // This finds when price crosses above longTrigger or below shortTrigger
    const triggerResult = scanForTrigger(state);
    
    // If no trigger found, we're done (no more trades possible)
    if (!triggerResult) {
      state.log(`  No trigger crossed from candle[${state.scanStartIndex}] onwards. Done.`);
      break;
    }

    // Log trade discovery - trigger found (Open row)
    if (state.enableLogging) {
      logMain.debug("trade discovered - trigger", {
        trade: {
          number: state.tradeNumber,
          sessionCount: state.sessionTradeCount,
        },
        trigger: {
          candleIndex: triggerResult.candleIndex,
          direction: triggerResult.direction,
          candle: state.candles[triggerResult.candleIndex],
        },
        priceState: {
          currentStartPrice: state.currentStartPrice,
          longTrigger: state.longTrigger,
          shortTrigger: state.shortTrigger,
          hasSwitched: state.hasSwitched,
        },
        alternatingLogic: {
          lastTradeDirection: state.lastTradeDirection,
          lastExitNote: state.lastExitNote,
          lastExitWasSameCandleEligible: state.lastExitWasSameCandleEligible,
        },
        resetState: {
          hasResetAt935: state.hasResetAt935,
          candle935Index: state.candle935Index,
          priceAt935: state.priceAt935,
        },
      }, undefined, "trade-disc-open-001", ["trade-discovery", "trigger-found"]);
    }

    // Calculate entry price and timestamp
    const entry = calculateEntry(triggerResult, state);
    
    // Calculate TP and SL prices
    // These are based on currentStartPrice (which may have shifted at 9:35)
    const { tpPrice, slPrice } = calculateTpAndSl(triggerResult.direction, triggerResult.candleIndex, state);
    
    // Scan for exit condition (TP hit, SL hit, or EOD)
    const exitResult = scanForExit(
      triggerResult.candleIndex,
      entry.price,
      entry.timestamp,
      triggerResult.direction,
      tpPrice,
      slPrice,
      state
    );
    
    // Log trade discovery - exit found (Close row)
    if (state.enableLogging) {
      logMain.debug("trade discovered - exit", {
        trade: {
          number: state.tradeNumber,
          sessionCount: state.sessionTradeCount,
        },
        exit: {
          candleIndex: exitResult.candleIndex,
          price: exitResult.price,
          timestamp: exitResult.timestamp,
          note: exitResult.note,
          slippage: exitResult.slippage,
        },
        priceState: {
          currentStartPrice: state.currentStartPrice,
          longTrigger: state.longTrigger,
          shortTrigger: state.shortTrigger,
          hasSwitched: state.hasSwitched,
        },
        alternatingLogic: {
          lastTradeDirection: state.lastTradeDirection,
          lastExitNote: state.lastExitNote,
          lastExitWasSameCandleEligible: state.lastExitWasSameCandleEligible,
        },
        resetState: {
          hasResetAt935: state.hasResetAt935,
          candle935Index: state.candle935Index,
          priceAt935: state.priceAt935,
        },
      }, undefined, "trade-disc-close-001", ["trade-discovery", "exit-found"]);
    }
    
    // Build trade rows (Open + Close) and add to results
    const { openRow, closeRow } = buildTradeRows(
      state.tradeNumber,
      triggerResult.direction,
      entry.price,
      entry.timestamp,
      exitResult,
      state
    );
    state.allRows.push(openRow, closeRow);

    // Update candle decision logs with actual trade rows created
    // Note: This is a simplified approach - in a full implementation, we'd track decisions per candle
    // and update them. For now, we log decisions during scans and note trades will be created.
    if (state.enableLogging) {
      // Log that trades were created for trigger candle
      state.logTrigger.debug("trades created for candle", {
        triggerCandleIndex: triggerResult.candleIndex,
        exitCandleIndex: exitResult.candleIndex,
        trades: [openRow, closeRow],
      }, "candleDecision", "candle-trades-created-001", ["trade-created"]);
    }
    
    // Update alternating logic state for next trade
    state.lastTradeDirection = triggerResult.direction;
    state.lastExitNote = exitResult.note;
    
    // If we hit EOD or forced close, we're done
    if (exitResult.note === "EOD Push" || exitResult.note === "Forced 9:35 Close") {
      if (exitResult.note === "Forced 9:35 Close") {
        // Reset session count after forced close
        state.sessionTradeCount = 0;
        state.lastTradeDirection = null;
        state.lastExitNote = null; // Reset exit note to clear alternating logic
        state.scanStartIndex = exitResult.candleIndex; // Start next scan AT 9:35
        state.log(`  Skipping re-eligibility due to Forced Close. Next scan starts at ${state.scanStartIndex} (9:35 candle)`);
        continue; // Skip re-eligibility check, restart loop
      } else {
        state.log(`  EOD reached, no more trades possible`);
        break;
      }
    }
    
    // Scan for re-eligibility (price must return between triggers)
    // After a trade exits, price must return to the zone between triggers before
    // another trade can be triggered. This prevents immediate re-entry.
    const reEligibleIndex = scanForReeligibility(exitResult.candleIndex, state);
    
    // If no re-eligibility found, we're done
    if (reEligibleIndex === -1) {
      state.log(`  No re-eligibility found - price never returned between triggers. Done.`);
      state.lastExitWasSameCandleEligible = false; // Reset flag
      break;
    }
    
    // Start next trade scan from after the re-eligible candle
    state.scanStartIndex = reEligibleIndex + 1;
    state.log(`  Next scan will start from candle[${state.scanStartIndex}] (after re-eligibility)`);
  }

  state.log("=".repeat(70));
  state.log(`GENERATION COMPLETE: ${state.allRows.length} rows (${state.tradeNumber} round trips)`);
  state.log("=".repeat(70));

  return state.allRows;
}

/**
 * Find the index of the 9:35:00 candle (first candle at or after 9:35:00).
 * 
 * **What this function does:**
 * Scans through the candles array to find the first candle that contains "09:35:00"
 * in its timestamp. This is where the start price may shift from the 9:30 open
 * to the 9:35 open.
 * 
 * **Time Format Handling:**
 * We check for both "09:35:00" and "T09:35:00" to handle different timestamp
 * formats (ISO format uses "T" separator, e.g., "2024-01-01T09:35:00").
 * 
 * **Returns:**
 * - Index of 9:35 candle if found
 * - -1 if not found (means no 9:35 transition for this day)
 * 
 * @param candles - Array of candles to search
 * @returns Index of 9:35:00 candle, or -1 if not found
 */
function find935CandleIndex(candles: Candle[]): number {
  // Target: 9:35:00 AM ET (which is 14:35:00 UTC in standard time, or 13:35:00 UTC in DST)
  // We'll match any candle with time containing "09:35:00" or "T09:35:00"
  for (let i = 0; i < candles.length; i++) {
    const candle = candles[i];
    const timeStr = candle.time;
    // Match 9:35:00 in various timezone formats (ET, UTC, etc.)
    // Look for "09:35:00" or "T09:35:00" in the timestamp
    if (timeStr.includes("09:35:00") || timeStr.includes("T09:35:00")) {
      return i;
    }
  }
  return -1;
}

/**
 * Check if we need to reset session state at 9:35.
 * 
 * **What this function does:**
 * If we've crossed the 9:35 candle index without a trade being active, we need
 * to reset the session state. This includes:
 * - Resetting sessionTradeCount to 0 (allowing new trades in post-9:35 session)
 * - Resetting lastTradeDirection (clearing alternating logic)
 * - Updating currentStartPrice to the 9:35 open price
 * - Recalculating triggers based on new start price
 * 
 * **When does this happen?**
 * This can happen in two scenarios:
 * 1. Before checking limits: We're waiting for a trigger and cross 9:35
 * 2. During trigger scan: We're scanning for a trigger and cross 9:35
 * 
 * **State Modification:**
 * This function modifies the state object directly (objects are passed by reference
 * in JavaScript/TypeScript, so changes are visible to the caller).
 * 
 * @param state - Trading state object (will be modified)
 */
function check935Reset(state: TradingState): void {
  // If we passed 9:35 while waiting for next trade, reset session state
  if (!state.hasResetAt935 && state.candle935Index >= 0 && state.scanStartIndex >= state.candle935Index) {
    state.hasResetAt935 = true;
    state.sessionTradeCount = 0;
    state.lastTradeDirection = null; // Reset eligibility
    state.lastExitNote = null; // Reset exit note to clear alternating logic
    state.lastExitWasSameCandleEligible = false; // Reset same-candle flag
    state.log(`*** 9:35 RESET TRIGGERED (No trade active) ***`);
    
    // Ensure start price is updated if not already
    if (!state.hasSwitched && state.priceAt935 != null) {
      state.currentStartPrice = state.priceAt935;
      state.hasSwitched = true;
      // Recalculate triggers with new start price
      state.longTrigger = state.currentStartPrice + state.triggerPts;
      state.shortTrigger = state.currentStartPrice - state.triggerPts;
      state.log(`  Start price switched to ${state.currentStartPrice} (at Reset)`);
    }
  }
}

/**
 * Extract MM:SS from a timestamp string.
 * 
 * @param timestamp - Timestamp string (e.g., "2024-01-01T09:35:21" or "09:35:21")
 * @returns MM:SS format (e.g., "35:21") or empty string if not found
 */
function extractTimeMMSS(timestamp: string): string {
  const timeMatch = timestamp.match(/(\d{2}):(\d{2})(?::(\d{2}))?/);
  if (!timeMatch) return "";
  const [, hour, minute, second] = timeMatch;
  return `${minute}:${second || "00"}`;
}

/**
 * Extract minute:second:milliseconds from timestamp string.
 * Handles formats like "2025-12-03T09:35:29-05:00" or "2025-12-03T09:35:29.123-05:00"
 * Returns format: "35:29.123" or "35:29.000" if no milliseconds
 */
function extractTimeMMSSMS(timestamp: string): string {
  // Match HH:MM:SS.mmm or HH:MM:SS
  const timeMatch = timestamp.match(/(\d{2}):(\d{2}):(\d{2})(?:\.(\d{1,3}))?/);
  if (!timeMatch) return "";
  const [, hour, minute, second, millisecond] = timeMatch;
  const ms = millisecond ? millisecond.padEnd(3, '0').substring(0, 3) : "000";
  return `${minute}:${second}.${ms}`;
}

/**
 * Check if a timestamp is before the cutoff time.
 * 
 * **What this function does:**
 * Parses a timestamp and compares it to the cutoff time to determine if trades
 * are still allowed. This prevents new entries after a certain time (e.g., 9:45 AM).
 * 
 * **Time Parsing:**
 * Handles various timestamp formats by extracting HH:MM:SS or HH:MM portions
 * and normalizing them for comparison.
 * 
 * **Comparison Logic:**
 * Uses `<` (less than) to prevent trades at or after cutoff time. This means
 * if cutoff is "09:45", trades are allowed up to but not including 9:45:00.
 * 
 * @param timestamp - Timestamp string or number to check
 * @param cutoffTime - Cutoff time in HH:MM format (e.g., "09:45")
 * @returns True if timestamp is before cutoff time, false otherwise
 */
function isBeforeCutoff(timestamp: string | number, cutoffTime: string): boolean {
  if (!timestamp) return true;
  const timeStr = typeof timestamp === "string" ? timestamp : new Date(timestamp).toISOString();
  // Extract time portion (HH:MM:SS or HH:MM)
  const timeMatch = timeStr.match(/(\d{2}):(\d{2})(?::(\d{2}))?/);
  if (!timeMatch) return true;
  const [, hour, minute, second] = timeMatch;
  // Normalize to HH:MM:SS format for comparison
  const candleTime = `${hour}:${minute}:${second || '00'}`;
  // Normalize cutoff time to HH:MM:SS format
  const cutoffMatch = cutoffTime.match(/(\d{2}):(\d{2})(?::(\d{2}))?/);
  if (!cutoffMatch) return true;
  const [, cutoffHour, cutoffMinute, cutoffSecond] = cutoffMatch;
  const cutoffTimeNormalized = `${cutoffHour}:${cutoffMinute}:${cutoffSecond || '00'}`;
  // Use < to prevent trades at or after cutoff time
  return candleTime < cutoffTimeNormalized;
}

/**
 * Scan candles for trigger crossing (entry signal).
 * 
 * **What this function does:**
 * Scans candles starting from scanStartIndex to find when price crosses above
 * longTrigger or below shortTrigger. This is the entry signal for a new trade.
 * 
 * **9:35 Transition Handling:**
 * If we cross the 9:35 candle during the scan, we:
 * - Reset session state (allowing new trades in post-9:35 session)
 * - Update currentStartPrice to 9:35 open
 * - Recalculate triggers based on new start price
 * 
 * **Alternating Logic:**
 * After an SL or TP hit, we must trade the opposite direction. This function enforces
 * that by blocking triggers in the wrong direction.
 * 
 * **Tiebreaker Logic:**
 * If both triggers are hit on the same candle, we choose the one closer to
 * the candle's open price (more likely to have been hit first).
 * 
 * @param state - Trading state object (will be modified)
 * @returns Trigger result if found, null otherwise
 */
function scanForTrigger(state: TradingState): TriggerResult | null {
  let triggerCandleIndex = -1;
  let direction: "long" | "short" | null = null;

  // Determine if we need to force opposite direction after SL or TP
  // BUT: If last exit was same-candle with eligible price, allow either direction (override alternating logic)
  let mustAlternate = (state.lastExitNote === "SL Hit" || state.lastExitNote === "PT Hit") && state.lastTradeDirection !== null && !state.lastExitWasSameCandleEligible;
  let requiredDirection = mustAlternate 
    ? (state.lastTradeDirection === "long" ? "short" : "long")
    : null;
  
  if (mustAlternate && requiredDirection) {
    if (state.enableLogging) {
      /**
       * Logs when alternating logic is active (forces opposite direction after SL or TP).
       * DEBUG level: Decision point - shows why direction is being forced.
       * 
       * @param message - "alternating logic active"
       * @param data - Trade number, last trade direction, last exit note, and required direction
       * @param subfunction - undefined (no subfunction)
       * @param id - "12dot10altactive" - Custom ID for easy code lookup
       */
      state.logAlternate.debug("alternating logic active", {
        tradeNumber: state.tradeNumber,
        lastTradeDirection: state.lastTradeDirection,
        lastExitNote: state.lastExitNote,
        requiredDirection,
      }, undefined, "12dot10altactive");
    }
    state.log(`  *** ALTERNATING LOGIC: Last trade was ${state.lastTradeDirection} with ${state.lastExitNote}, must trade ${requiredDirection} next ***`);
  }

  // Scan candles starting from scanStartIndex
  for (let i = state.scanStartIndex; i < state.candles.length; i++) {
    const candle = state.candles[i];
    
    // Check cutoff time - no trades after cutoff
    if (!isBeforeCutoff(candle.time, state.cutoffTime)) {
      state.log(`  Candle[${i}] is after cutoff time ${state.cutoffTime}, stopping trade search`);
      break;
    }
    
    // Check if we've reached 9:35 and need to switch start price
    if (!state.hasSwitched && state.candle935Index >= 0 && i >= state.candle935Index && state.priceAt935 != null) {
      
      // 9:35 Reset Logic during scan
      if (!state.hasResetAt935) {
        state.hasResetAt935 = true;
        state.sessionTradeCount = 1; // Reset to 1 (this attempt is #1 of new session)
        state.lastTradeDirection = null; // Reset eligibility
        state.lastExitNote = null; // Reset exit note to clear alternating logic
        
        // Clear alternating logic because we reset!
        mustAlternate = false;
        requiredDirection = null;

        state.log(`*** 9:35 RESET TRIGGERED (During Trigger Scan) ***`);
      }
      
      state.currentStartPrice = state.priceAt935;
      state.hasSwitched = true;
      // Recalculate triggers with new start price
      state.longTrigger = state.currentStartPrice + state.triggerPts;
      state.shortTrigger = state.currentStartPrice - state.triggerPts;
      state.log(`  *** 9:35 TRANSITION at candle[${i}] ***`);
      state.log(`  Start price switched: ${state.startPrice} -> ${state.currentStartPrice}`);
      state.log(`  Triggers recalculated:`);
      state.log(`    Long Trigger  = ${state.currentStartPrice} + ${state.triggerPts} = ${state.longTrigger}`);
      state.log(`    Short Trigger = ${state.currentStartPrice} - ${state.triggerPts} = ${state.shortTrigger}`);
    }
    
    // Check if triggers are hit
    const longHit = candle.high >= state.longTrigger;
    const shortHit = candle.low <= state.shortTrigger;
    
    // Apply alternating logic: if we must alternate, only accept the required direction
    let allowedLongHit = longHit;
    let allowedShortHit = shortHit;
    
    if (mustAlternate && requiredDirection) {
      if (requiredDirection === "long") {
        allowedShortHit = false; // Only allow long
        if (state.enableLogging) {
          /**
           * Logs when alternating logic blocks short trigger (forces long only).
           * TRACE level: Hotspot/loop - per-candle logging during trigger scan.
           * 
           * @param message - "blocking short"
           * @param data - Candle index and required direction
           * @param subfunction - undefined (no subfunction)
           * @param id - "12dot10blockshort" - Custom ID for easy code lookup
           */
          state.logAlternate.trace("blocking short", { candleIndex: i, requiredDirection }, undefined, "12dot10blockshort");
        }
        state.log(`  Candle[${i}]: Alternating logic - only allowing LONG trigger`);
      } else {
        allowedLongHit = false; // Only allow short
        if (state.enableLogging) {
          /**
           * Logs when alternating logic blocks long trigger (forces short only).
           * TRACE level: Hotspot/loop - per-candle logging during trigger scan.
           * 
           * @param message - "blocking long"
           * @param data - Candle index and required direction
           * @param subfunction - undefined (no subfunction)
           * @param id - "12dot10blocklong" - Custom ID for easy code lookup
           */
          state.logAlternate.trace("blocking long", { candleIndex: i, requiredDirection }, undefined, "12dot10blocklong");
        }
        state.log(`  Candle[${i}]: Alternating logic - only allowing SHORT trigger`);
      }
    }

    // Build CandleDecision for this candle (trigger-scan phase)
    if (state.enableLogging) {
      const reset935Triggered = !state.hasSwitched && state.candle935Index >= 0 && i >= state.candle935Index && state.priceAt935 != null;
      
      const decision: CandleDecision = {
        candleIndex: i,
        candle: candle,
        // Position tracking fields (at top of JSON as requested)
        entryPrice: null, // No position during trigger-scan phase
        currentPrice: null, // No position during trigger-scan phase
        unrealizedEquity: null, // No position during trigger-scan phase
        checks: {
          cutoffTime: {
            passed: isBeforeCutoff(candle.time, state.cutoffTime),
            value: candle.time,
            cutoff: state.cutoffTime,
          },
          triggerHit: {
            long: longHit,
            short: shortHit,
            levels: {
              long: state.longTrigger,
              short: state.shortTrigger,
            },
          },
          alternatingLogic: {
            active: mustAlternate,
            requiredDirection: requiredDirection as "long" | "short" | null,
          },
          reset935: {
            triggered: reset935Triggered,
            newStartPrice: reset935Triggered ? state.priceAt935 : null,
          },
        },
        conditionalFlags: {
          mustAlternate,
          allowedLongHit,
          allowedShortHit,
        },
        outcome: {
          decision: (allowedLongHit || allowedShortHit) ? "TRADE_FOUND" : "NO_TRADE",
          reason: (allowedLongHit || allowedShortHit)
            ? `Trigger hit: ${allowedLongHit && allowedShortHit ? "both" : (allowedLongHit ? "long" : "short")}`
            : "No trigger crossed",
          direction: (allowedLongHit || allowedShortHit) 
            ? (allowedLongHit && allowedShortHit 
                ? (Math.abs(candle.open - state.longTrigger) <= Math.abs(candle.open - state.shortTrigger) ? "long" : "short")
                : (allowedLongHit ? "long" : "short"))
            : undefined,
        },
        results: {
          tradesCreated: [], // Will be updated if trigger found - actual trades logged after buildTradeRows
        },
      };

      const tags = [
        "candle-decision",
        decision.outcome.decision === "TRADE_FOUND" ? "trade" : "no-trade",
        decision.checks.triggerHit.long || decision.checks.triggerHit.short ? "trigger-hit" : null,
        decision.checks.reset935.triggered ? "reset-935" : null,
        decision.checks.alternatingLogic.active ? "alternating-logic" : null,
      ].filter(Boolean) as string[];

      const timeStr = extractTimeMMSS(candle.time);
      state.logTrigger.debug(
        `candle decision ${timeStr}`,
        { decision },
        "candleDecision",
        "candle-decision-001",
        tags
      );
    }

    if (allowedLongHit || allowedShortHit) {
      triggerCandleIndex = i;
      if (allowedLongHit && allowedShortHit) {
        // Tiebreaker: which trigger is closer to open
        const distToLong = Math.abs(candle.open - state.longTrigger);
        const distToShort = Math.abs(candle.open - state.shortTrigger);
        direction = distToLong <= distToShort ? "long" : "short";
        if (state.enableLogging) {
          /**
           * Logs when both triggers hit on same candle (tiebreaker logic).
           * DEBUG level: Decision point - shows which direction was chosen and why.
           * 
           * @param message - "both triggers hit"
           * @param data - Candle index, distances to each trigger, and chosen direction
           * @param subfunction - undefined (no subfunction)
           * @param id - "12dot10bothtrig" - Custom ID for easy code lookup
           */
          state.logTrigger.debug("both triggers hit", {
            candleIndex: i,
            distToLong,
            distToShort,
            direction,
          }, undefined, "12dot10bothtrig");
        }
        state.log(`  Candle[${i}] BOTH triggers hit! Tiebreaker -> ${direction}`);
      } else {
        direction = allowedLongHit ? "long" : "short";
        if (state.enableLogging) {
          /**
           * Logs when a single trigger is hit (long or short, not both).
           * DEBUG level: Decision point - shows which trigger was hit.
           * 
           * @param message - "trigger hit"
           * @param data - Candle index and direction (long or short)
           * @param subfunction - undefined (no subfunction)
           * @param id - "12dot10trighit" - Custom ID for easy code lookup
           */
          state.logTrigger.debug("trigger hit", {
            candleIndex: i,
            direction,
          }, undefined, "12dot10trighit");
        }
        state.log(`  Candle[${i}] ${direction.toUpperCase()} trigger hit`);
      }
      break;
    }
  }

  // Return trigger result if found, null otherwise
  if (triggerCandleIndex === -1 || !direction) {
    return null;
  }

  return {
    candleIndex: triggerCandleIndex,
    direction,
  };
}

/**
 * Calculate entry price and timestamp from trigger result.
 * 
 * **What this function does:**
 * Takes a trigger result (candle index and direction) and calculates:
 * - Entry price: trigger price + entry slippage
 * - Entry timestamp: timestamp of the trigger candle
 * 
 * **Slippage:**
 * Entry slippage accounts for the fact that we don't get filled at exactly
 * the trigger price. For longs, we add slippage (fill slightly above trigger).
 * For shorts, we subtract slippage (fill slightly below trigger).
 * 
 * @param triggerResult - Trigger result from scanForTrigger()
 * @param state - Trading state object
 * @returns Entry price and timestamp
 */
function calculateEntry(
  triggerResult: TriggerResult,
  state: TradingState
): { price: number; timestamp: string } {
  const entrySlippage = calculateEntrySlippage(triggerResult.direction);
  const triggerPrice = triggerResult.direction === "long" 
    ? state.longTrigger 
    : state.shortTrigger;
  const entryPrice = triggerResult.direction === "long" 
    ? triggerPrice + entrySlippage 
    : triggerPrice - entrySlippage;
  const entryTimestamp = state.candles[triggerResult.candleIndex].time;

  state.log(`  Entry: ${entryPrice} at ${entryTimestamp}`);
  state.log(`  Entry calculation: ${triggerResult.direction} trigger (${triggerPrice}) + slippage (${entrySlippage}) = ${entryPrice}`);

  return {
    price: entryPrice,
    timestamp: entryTimestamp,
  };
}

/**
 * Calculate TP and SL prices based on current start price.
 * 
 * **What this function does:**
 * Calculates take profit (TP) and stop loss (SL) prices based on:
 * - Direction (long or short)
 * - Current start price (may have shifted at 9:35)
 * - Profit target points (for TP)
 * - Stop loss points (for initial SL)
 * - Trigger prices (for SL calculation)
 * 
 * **9:35 Transition Handling:**
 * If we haven't switched yet but the trigger candle is at or after 9:35, we update
 * the start price and recalculate triggers before calculating TP/SL.
 * 
 * **TP Calculation:**
 * - Long: TP = currentStartPrice + profitTargetPts
 * - Short: TP = currentStartPrice - profitTargetPts
 * 
 * **SL Calculation:**
 * - Long: SL = longTrigger - stopLossPts
 * - Short: SL = shortTrigger + stopLossPts
 * 
 * @param direction - Trade direction (long or short)
 * @param triggerCandleIndex - Index of candle where trigger was hit
 * @param state - Trading state object (will be modified)
 * @returns TP price and SL price
 */
function calculateTpAndSl(
  direction: "long" | "short",
  triggerCandleIndex: number,
  state: TradingState
): { tpPrice: number; slPrice: number } {
  // Ensure we've switched if we're past 9:35
  // Only switch if trigger candle is at or after 9:35 candle
  if (!state.hasSwitched && state.candle935Index >= 0 && triggerCandleIndex >= state.candle935Index && state.priceAt935 != null) {
    state.currentStartPrice = state.priceAt935;
    state.hasSwitched = true;
    state.longTrigger = state.currentStartPrice + state.triggerPts;
    state.shortTrigger = state.currentStartPrice - state.triggerPts;
    state.log(`  *** 9:35 TRANSITION detected during TP/SL calc ***`);
    state.log(`  Start price switched: ${state.startPrice} -> ${state.currentStartPrice}`);
  }
  
  let tpPrice: number;
  let slPrice: number;

  if (direction === "long") {
    tpPrice = state.currentStartPrice + state.profitTargetPts;
    slPrice = state.longTrigger - state.stopLossPts;
  } else {
    tpPrice = state.currentStartPrice - state.profitTargetPts;
    slPrice = state.shortTrigger + state.stopLossPts;
  }

  state.log(`  TP: ${tpPrice} (based on currentStartPrice: ${state.currentStartPrice}), SL: ${slPrice}`);

  return { tpPrice, slPrice };
}

/**
 * Scan candles for exit condition (TP hit, SL hit, or EOD).
 * 
 * **What this function does:**
 * Scans candles starting from trigger candle to find when the trade should exit.
 * Exit conditions:
 * - TP hit: Price reaches take profit target
 * - SL hit: Price reaches stop loss
 * - EOD Push: End of day reached, close at entry price (net=0)
 * - Forced 9:35 Close: Trade must close at 9:35 for session reset
 * 
 * **Intrabar Resolution:**
 * For same-candle exits (exit on trigger candle), we use intrabar resolution
 * to properly handle the order of events (entry vs exit) within the candle.
 * 
 * @param triggerCandleIndex - Index of candle where trigger was hit
 * @param entryPrice - Entry price of the trade
 * @param entryTimestamp - Entry timestamp
 * @param direction - Trade direction (long or short)
 * @param tpPrice - Take profit target price
 * @param initialSlPrice - Initial stop loss price (may be adjusted by stepped SL logic)
 * @param state - Trading state object (will be modified)
 * @returns Exit result with price, timestamp, reason, and slippage
 */
function scanForExit(
  triggerCandleIndex: number,
  entryPrice: number,
  entryTimestamp: string,
  direction: "long" | "short",
  tpPrice: number,
  initialSlPrice: number,
  state: TradingState
): ExitResult {
  // Stepped SL state tracking
  let currentSlPrice = initialSlPrice;
  let highestProfitPts = 0; // For longs: best unrealized profit reached
  let lowestProfitPts = 0; // For shorts: best unrealized profit reached (negative)
  
  // Parse and sort thresholds if stepped SL config is provided
  // Thresholds are sorted ascending so we check them in order
  const thresholds: number[] = state.stopLossByProfitThreshold 
    ? Object.keys(state.stopLossByProfitThreshold)
        .map(k => Number(k))
        .filter(k => !isNaN(k))
        .sort((a, b) => a - b)
    : [];
  
  if (state.enableLogging) {
    state.logSteppedSL.debug("stepped sl config", {
      config: state.stopLossByProfitThreshold,
      thresholds,
    }, undefined, "12dot10dot1-sl-config");
  }
  
  // Also log to console for debugging (even if enableLogging is false, we want to see stepped SL config)
  if (state.stopLossByProfitThreshold) {
    state.log(`  Stepped SL config loaded: ${JSON.stringify(state.stopLossByProfitThreshold)}, thresholds: [${thresholds.join(", ")}]`);
  } else {
    state.log(`  No stepped SL config provided`);
  }
  
  // Track SL changes for UI visualization (timestamp -> new SL price)
  const slChanges: Array<{ timestamp: string; slPrice: number; profitLocked: number }> = [];
  
  let exitCandleIndex = -1;
  let exitNote: "PT Hit" | "SL Hit" | "EOD Push" | "Forced 9:35 Close" | string = "EOD Push";
  let exitPrice = entryPrice; // Default: EOD Push closes at entry price (net=0)
  let exitTimestamp = state.candles[state.candles.length - 1].time;
  let exitSlippage: number | undefined;
  let profitLocked: number | undefined;
  let maxProfitReached: number | undefined;

  // Scan candles starting from trigger candle (not +1) to detect same-candle exits
  for (let i = triggerCandleIndex; i < state.candles.length; i++) {
    const candle = state.candles[i];
    let tpHit = false;
    let slHit = false;

    // Check for Forced Close at 9:35
    // If we hit 9:35 candle and haven't reset yet, force close the trade
    if (!state.hasResetAt935 && state.candle935Index >= 0 && i === state.candle935Index) {
      exitCandleIndex = i;
      exitTimestamp = candle.time;
      exitPrice = state.candles[i-1].close; // Close of 9:34
      exitNote = "Forced 9:35 Close";
      state.log(`  *** FORCED CLOSE at 9:35 ***`);
      
      state.hasResetAt935 = true;
      break;
    }

    // STEPPED SL LOGIC: Check unrealized profit and update SL if threshold crossed
    // IMPORTANT: This must run for ALL trades, including those starting after 9:35
    if (state.stopLossByProfitThreshold && thresholds.length > 0) {
      // Calculate unrealized profit for this candle
      // Profit is calculated from ENTRY PRICE - this is the standard way to measure unrealized profit
      // The thresholds represent profit from entry, not from start price
      let unrealizedProfitPts: number;
      if (direction === "long") {
        // For longs: profit = highest price reached - entry price
        const candleHighProfit = candle.high - entryPrice;
        highestProfitPts = Math.max(highestProfitPts, candleHighProfit);
        unrealizedProfitPts = highestProfitPts;
      } else {
        // For shorts: profit = entry price - lowest price reached
        const candleLowProfit = entryPrice - candle.low;
        lowestProfitPts = Math.max(lowestProfitPts, candleLowProfit);
        unrealizedProfitPts = lowestProfitPts;
      }
      
      if (state.enableLogging) {
        state.logSteppedSL.trace("checking profit", {
          candleIndex: i,
          time: candle.time,
          candle: {
            open: candle.open,
            high: candle.high,
            low: candle.low,
            close: candle.close,
          },
          entryPrice,
          direction,
          unrealizedProfitPts,
          currentSlPrice,
        }, undefined, "12dot10dot1-check-profit");
      }
      
      // Log profit check to console for debugging (log all checks, not just high profit)
      // This helps debug why stepped SL might not be working for certain trades
      if (unrealizedProfitPts >= 5 || i === triggerCandleIndex) {
        state.log(`  Stepped SL check at ${candle.time}: ${direction} trade, unrealized profit: ${unrealizedProfitPts.toFixed(2)}pts, current SL: ${currentSlPrice.toFixed(2)}, entry: ${entryPrice.toFixed(2)}`);
      }
      
      // Check each threshold (sorted ascending) to see if we should update SL
      for (const threshold of thresholds) {
        const thresholdCrossed = unrealizedProfitPts >= threshold;
        
        if (thresholdCrossed) {
          const profitToLock = state.stopLossByProfitThreshold![String(threshold)];
          if (profitToLock != null) {
            // Calculate new SL price based on profit to lock
            let newSlPrice: number;
            if (direction === "long") {
              // For longs: SL = Entry + profitToLock
              newSlPrice = entryPrice + profitToLock;
            } else {
              // For shorts: SL = Entry - profitToLock
              newSlPrice = entryPrice - profitToLock;
            }
            
            // Only update if new SL is better (higher for longs, lower for shorts)
            const shouldUpdate = direction === "long" 
              ? newSlPrice > currentSlPrice 
              : newSlPrice < currentSlPrice;
            
            if (state.enableLogging) {
              state.logSteppedSL.trace("threshold crossed", {
                threshold,
                profitToLock,
                newSlPrice,
                currentSlPrice,
                shouldUpdate,
                unrealizedProfitPts,
                entryPrice,
                direction,
              }, undefined, "12dot10dot1-threshold");
            }
            
            // Always log threshold crossings for debugging (even if enableLogging is false)
            state.log(`  Stepped SL: threshold ${threshold}pts crossed (profit: ${unrealizedProfitPts.toFixed(2)}pts), would lock ${profitToLock}pts. Current SL: ${currentSlPrice.toFixed(2)}, new SL: ${newSlPrice.toFixed(2)}, shouldUpdate: ${shouldUpdate}`);
            
            if (shouldUpdate) {
              // Extract time for log title (minute:second)
              const timeStr = extractTimeMMSS(candle.time);
              
              if (state.enableLogging) {
                state.logSteppedSL.debug(`SL adjusted ${timeStr}`, {
                  candleIndex: i,
                  candleTime: candle.time,
                  oldSlPrice: currentSlPrice,
                  newSlPrice,
                  profitLocked: profitToLock,
                  threshold,
                  direction,
                }, undefined, "12dot10dot1-update-sl");
              }
              // Log to console for debugging (even if enableLogging is false)
              state.log(`  *** STEPPED SL UPDATE: ${direction} trade at ${candle.time}, threshold ${threshold}pts crossed, locking ${profitToLock}pts profit. SL: ${currentSlPrice.toFixed(2)} -> ${newSlPrice.toFixed(2)} ***`);
              currentSlPrice = newSlPrice;
              
              // Log "SL adjusted" with time in title (min:sec) - always log even if enableLogging is false
              state.log(`SL adjusted ${timeStr || "unknown"}`);
              
              slChanges.push({
                timestamp: candle.time,
                slPrice: currentSlPrice,
                profitLocked: profitToLock,
              });
            } else {
              // Log when threshold is crossed but SL isn't updated (shouldn't happen often)
              state.log(`  Stepped SL: threshold ${threshold}pts crossed but SL not updated (current: ${currentSlPrice.toFixed(2)}, new: ${newSlPrice.toFixed(2)}, direction: ${direction})`);
            }
          } else {
            state.log(`  Stepped SL: threshold ${threshold}pts crossed but profitToLock is null/undefined`);
          }
        }
      }
    }

    // For same-candle scenarios (i === triggerCandleIndex), use intrabar resolution
    // to properly handle path order (stop hit before entry doesn't block entry)
    if (i === triggerCandleIndex) {
      const entryLevel = direction === "long" ? state.longTrigger : state.shortTrigger;
      const intrabarResult = resolveIntrabarDirectional({
        candle: { O: candle.open, H: candle.high, L: candle.low, C: candle.close, V: candle.volume || 0 },
        side: direction,
        entryLevel: entryLevel,
        stopLevel: currentSlPrice, // Use current SL (may have been updated by stepped logic)
        targetLevel: tpPrice,
      });

      // Use intrabar resolution result for same-candle exits
      if (intrabarResult.event === "FILL_STOP_SAME_BAR") {
        exitCandleIndex = i;
        exitTimestamp = candle.time;
        
        // Check if SL was adjusted (stepped SL)
        const slWasAdjusted = currentSlPrice !== initialSlPrice;
        if (slWasAdjusted) {
          // Calculate profit locked and max profit reached
          profitLocked = direction === "long" 
            ? currentSlPrice - entryPrice 
            : entryPrice - currentSlPrice;
          maxProfitReached = direction === "long" 
            ? highestProfitPts 
            : lowestProfitPts;
          
          // Create enhanced note with profit information
          exitNote = `hit adjusted sl of ${currentSlPrice.toFixed(2)} based on profit of ${profitLocked.toFixed(2)}`;
          
          if (state.enableLogging) {
            /**
             * Logs when adjusted stop loss is hit on same candle (stepped SL).
             * INFO level: Major milestone - adjusted SL hit with profit locked.
             * 
             * @param message - "hit adjusted sl"
             * @param data - Candle index, exit price, adjusted SL price, profit locked, max profit reached
             * @param id - "12dot10dot1-adj-sl-hit-same" - Custom ID for easy code lookup
             */
            state.logSteppedSL.info("hit adjusted sl", {
              candleIndex: i,
              exitPrice: intrabarResult.exitFill!,
              adjustedSlPrice: currentSlPrice,
              initialSlPrice,
              profitLocked,
              maxProfitReached,
              entryPrice,
              sameCandle: true,
            }, "12dot10dot1-adj-sl-hit-same", ["hit adjusted sl"]);
          }

/**
 * Build Open and Close trade rows from trade data.
 * 
 * **What this function does:**
 * Takes trade data (entry, exit, direction) and creates two AlgoTradeRow objects:
 * - Open row: Entry timestamp, price, direction, slippage
 * - Close row: Exit timestamp, price, direction, exit reason, net P&L, slippage
 * 
 * **P&L Calculation:**
 * - Long: net = exitPrice - entryPrice
 * - Short: net = entryPrice - exitPrice
 * 
 * **Row IDs:**
 * Each row gets a unique ID based on trade number:
 * - Open: "open-{tradeNumber}"
 * - Close: "close-{tradeNumber}"
 * 
 * @param tradeNumber - Trade number (for row IDs)
 * @param direction - Trade direction (long or short)
 * @param entryPrice - Entry price
 * @param entryTimestamp - Entry timestamp
 * @param exitResult - Exit result from scanForExit()
 * @param state - Trading state object (will be modified)
 * @returns Open and Close AlgoTradeRow objects
 */
function buildTradeRows(
  tradeNumber: number,
  direction: "long" | "short",
  entryPrice: number,
  entryTimestamp: string,
  exitResult: ExitResult,
  state: TradingState
): { openRow: AlgoTradeRow; closeRow: AlgoTradeRow } {
  // Calculate net P&L
  const net = direction === "long" 
    ? exitResult.price - entryPrice 
    : entryPrice - exitResult.price;

  state.log(`  Net P&L: ${net}`);

  const openType = direction === "long" ? "Open Long" : "Open Short";
  const closeType = direction === "long" ? "Close Long" : "Close Short";

  const openRow: AlgoTradeRow = {
    id: `open-${tradeNumber}`,
    timestamp: entryTimestamp,
    type: openType,
    price: entryPrice,
    slippage: calculateEntrySlippage(direction),
  };

  // Use enhanced note format if profitLocked is present (indicating adjusted SL)
  const closeNote = exitResult.profitLocked != null 
    ? exitResult.note 
    : exitResult.note;

  const closeRow: AlgoTradeRow = {
    id: `close-${tradeNumber}`,
    timestamp: exitResult.timestamp,
    type: closeType,
    price: exitResult.price,
    slippage: exitResult.slippage,
    note: closeNote,
    net: net,
  };

  state.log(`  Stored last trade: ${direction} exited with ${exitResult.note}`);

  return { openRow, closeRow };
}

/**
 * Scan candles for re-eligibility (price must return between triggers).
 * 
 * **What this function does:**
 * After a trade exits, price must return to the zone between triggers before
 * another trade can be triggered. This prevents immediate re-entry when price
 * is still beyond a trigger.
 * 
 * **Re-eligibility Zone:**
 * Price is "between triggers" if: shortTrigger < price < longTrigger
 * 
 * **Same-Candle Eligibility:**
 * If the exit candle itself has price in the zone, we mark it as eligible
 * and override alternating logic for immediate reentry.
 * 
 * **9:35 Transition Handling:**
 * If we cross 9:35 during re-eligibility scan, we reset session state and
 * recalculate triggers.
 * 
 * **Returns:**
 * - Index of candle where re-eligibility was found
 * - -1 if re-eligibility not found (no more trades possible)
 * 
 * @param exitCandleIndex - Index of candle where trade exited
 * @param state - Trading state object (will be modified)
 * @returns Re-eligibility candle index, or -1 if not found
 */
function scanForReeligibility(
  exitCandleIndex: number,
  state: TradingState
): number {
  // Check exit candle itself for re-eligibility (same-candle exit fix)
  // When exit happens on the same candle as entry, or when exit candle has price
  // between triggers, we should check that candle for re-eligibility
  const exitCandle = state.candles[exitCandleIndex];
  if (state.enableLogging) {
    state.logReentry.debug("checking exit candle", {
      exitCandleIndex,
      exitCandle: {
        open: exitCandle.open,
        high: exitCandle.high,
        low: exitCandle.low,
        close: exitCandle.close,
      },
      shortTrigger: state.shortTrigger,
      longTrigger: state.longTrigger,
    }, undefined, "12dot10checkexit");
  }
  
  // Check if exit candle has price in zone (between triggers)
  const exitLowInZone = exitCandle.low > state.shortTrigger && exitCandle.low < state.longTrigger;
  const exitHighInZone = exitCandle.high > state.shortTrigger && exitCandle.high < state.longTrigger;
  const exitOpenInZone = exitCandle.open > state.shortTrigger && exitCandle.open < state.longTrigger;
  const exitCloseInZone = exitCandle.close > state.shortTrigger && exitCandle.close < state.longTrigger;
  
  if (exitLowInZone || exitHighInZone || exitOpenInZone || exitCloseInZone) {
    // Exit candle has price in zone - can check for reentry
    if (state.enableLogging) {
      state.logReentry.debug("re-eligibility on exit candle", {
        exitCandleIndex,
        exitLowInZone,
        exitHighInZone,
        exitOpenInZone,
        exitCloseInZone,
      }, undefined, "12dot10exiteligible");
    }
    state.log(`  Re-eligibility found on exit candle[${exitCandleIndex}] - price between triggers`);
    // Mark that exit candle was eligible (overrides alternating logic for immediate reentry)
    state.lastExitWasSameCandleEligible = true;
    const isSameCandle = (exitCandleIndex === state.scanStartIndex);
    if (isSameCandle) {
      state.log(`  Same-candle exit with eligible price - alternating logic will be overridden for next trade`);
    } else {
      state.log(`  Exit candle has eligible price - alternating logic will be overridden for next trade`);
    }
    // Return exit candle index (caller will set scanStartIndex to exitCandleIndex + 1)
    return exitCandleIndex;
  } else {
    if (state.enableLogging) {
      state.logReentry.debug("exit candle not eligible", { exitCandleIndex }, undefined, "12dot10exitnoteligible");
    }
    state.lastExitWasSameCandleEligible = false; // Reset flag
  }
  
  // Otherwise, proceed with normal re-eligibility scan from exitCandleIndex + 1
  for (let i = exitCandleIndex + 1; i < state.candles.length; i++) {
    const candle = state.candles[i];
    
    // Check cutoff time - no new entries after cutoff
    if (!isBeforeCutoff(candle.time, state.cutoffTime)) {
      state.log(`  Cutoff time ${state.cutoffTime} reached during re-eligibility scan, no more trades allowed`);
      break;
    }
    
    // Check if we need to switch start price during re-eligibility scan
    if (!state.hasSwitched && state.candle935Index >= 0 && i >= state.candle935Index && state.priceAt935 != null) {
      state.currentStartPrice = state.priceAt935;
      state.hasSwitched = true;
      
      // Reset logic during re-eligibility if 9:35 is crossed
      state.hasResetAt935 = true;
      state.sessionTradeCount = 0;
      state.lastTradeDirection = null;
      state.lastExitNote = null; // Reset exit note to clear alternating logic
      
      state.longTrigger = state.currentStartPrice + state.triggerPts;
      state.shortTrigger = state.currentStartPrice - state.triggerPts;
      state.log(`  *** 9:35 TRANSITION during re-eligibility scan ***`);
      state.log(`  Start price switched: ${state.startPrice} -> ${state.currentStartPrice}`);
      state.log(`  Triggers recalculated: Long=${state.longTrigger}, Short=${state.shortTrigger}`);
      state.log(`  Session reset triggered during re-eligibility`);
    }
    
    // Price is "between triggers" if the candle's range crosses the zone
    // i.e., at some point during the candle, price was between triggers
    // This means: candle.low < longTrigger AND candle.high > shortTrigger
    // AND the candle actually enters the zone (not just spanning both triggers)
    const lowInZone = candle.low > state.shortTrigger && candle.low < state.longTrigger;
    const highInZone = candle.high > state.shortTrigger && candle.high < state.longTrigger;
    const openInZone = candle.open > state.shortTrigger && candle.open < state.longTrigger;
    const closeInZone = candle.close > state.shortTrigger && candle.close < state.longTrigger;
    
    if (state.enableLogging) {
      state.logReentry.trace("checking reentry candle", {
        candleIndex: i,
        time: candle.time,
        candle: {
          open: candle.open,
          high: candle.high,
          low: candle.low,
          close: candle.close,
        },
        zone: { shortTrigger: state.shortTrigger, longTrigger: state.longTrigger },
        checks: { lowInZone, highInZone, openInZone, closeInZone },
      }, undefined, "12dot10checkreentry");
    }
    
    if (lowInZone || highInZone || openInZone || closeInZone) {
      if (state.enableLogging) {
        state.logReentry.debug("re-eligibility found", { candleIndex: i }, undefined, "12dot10reentryfound");
      }
      state.log(`  Re-eligibility found at candle[${i}] - price returned between triggers`);
      // Reset same-candle flag since this is a different-candle re-eligibility
      state.lastExitWasSameCandleEligible = false;
      return i;
    }
  }

  // No re-eligibility found
  if (state.enableLogging) {
    state.logReentry.debug("no re-eligibility found", undefined, undefined, "12dot10noreentry");
  }
  state.lastExitWasSameCandleEligible = false; // Reset flag
  return -1;
}
