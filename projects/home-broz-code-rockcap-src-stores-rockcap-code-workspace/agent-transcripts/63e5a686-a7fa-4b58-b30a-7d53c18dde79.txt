# BUG: 20251218-consecutive-long-trades-same-interval Consecutive Long Trades in Same Interval Violating Alternation Rule
Status: Pending  <!-- Pending | Ready | In Progress | Review | Done -->

## Bug Summary
Multiple trades recorded in the first interval (930 candle) are all in the same direction (longs), violating the business rule that requires trades to alternate. The trades did not pass the opposite figure, which is a condition for allowing consecutive trades in the same direction.

## Stakeholder Notes (raw)
- **Date**: July 27th
- **Issue**: Two trades recorded in the first interval (930 candle)
  - Both trades were in the same direction (longs)
- **Concern**: 
  - This should not be possible due to existing rules and business laws that require trades to alternate.
  - The trades did not pass the opposite figure, which is a condition for allowing consecutive trades in the same direction.
- **Action Required**: 
  - Conduct a deep trace to investigate how this occurred and determine the underlying cause.

**URL**: http://localhost:3000/backtest/2025-07-29?strategy=strategy12.10.1&pt=30&sl=10&trg=8&candleType=1s&cutoffTime=09%3A40

**Note**: Screenshot evidence shows three consecutive "Open Long" trades in the 09:30 candle (RT#1 at 09:30:05, RT#2 at 09:30:10, RT#3 at 09:30:14), not two as initially reported. The core issue remains the same: consecutive long trades without an opposite figure.

## Expected Behavior
- Trades should alternate between long and short directions.
- Consecutive trades in the same direction should only be allowed if the price passes the "opposite figure" (e.g., a short trigger level when in a long sequence, or vice versa).
- The strategy should enforce the alternation rule before opening a new trade in the same direction.

## Observed Behavior
- Multiple consecutive "Open Long" trades occurred within the same 09:30 candle interval:
  - RT#1: Open Long at 09:30:05 ET (Price: 23604.25)
  - RT#2: Open Long at 09:30:10 ET (Price: 23604.25)
  - RT#3: Open Long at 09:30:14 ET (Price: 23604.25)
- No "Open Short" trade occurred between these consecutive longs.
- The price did not cross any short trigger level between these trades, meaning the "opposite figure" condition was not met.

## Repro Steps
1. Navigate to: http://localhost:3000/backtest/2025-07-29?strategy=strategy12.10.1&pt=30&sl=10&trg=8&candleType=1s&cutoffTime=09%3A40

/**
 * Extracts the hour and minute from a timestamp string (e.g., "09:30:05" -> "09:30")
 */
function getCandleInterval(timestamp: string): string {
  const match = timestamp.match(/^(\d{2}:\d{2})/);
  return match ? match[1] : "";
}

/**
 * Checks if a trade type indicates a long direction
 */
function isLongTrade(tradeType: string): boolean {
  return tradeType.includes("Long");
}

/**
 * Checks if a trade type indicates a short direction
 */
function isShortTrade(tradeType: string): boolean {
  return tradeType.includes("Short");
}

/**
 * Checks if a trade is an open trade
 */
function isOpenTrade(tradeType: string): boolean {
  return tradeType.includes("Open");
}

test("consecutive trades in same direction within same interval should not occur without opposite figure", {
  annotation: {
    type: "description",
    description: "Reproduces bug where multiple consecutive long trades occur in the same 09:30 candle interval without an intervening short trade or opposite figure condition being met. This test should FAIL on the current codebase (red state)."
  }
}, async ({ page }) => {
  const consoleErrors: string[] = [];
  
  // Set up console error listener
  page.on("console", (msg) => {
    if (msg.type() === "error") {
      consoleErrors.push(msg.text());
    }
  });

  // Navigate to the specific backtest page that reproduces the bug
  const bugUrl = "/backtest/2025-07-29?strategy=strategy12.10.1&pt=30&sl=10&trg=8&candleType=1s&cutoffTime=09%3A40";
  await page.goto(bugUrl);
  
  // Wait for algo trades to load
  await waitForAlgoTrades(page, 4); // Expect at least 4 rows (2+ trades)

  // Get all algo trade rows
  const trades = await getAlgoTradeRows(page);
  
  // Filter to only open trades
  const openTrades = trades.filter(trade => isOpenTrade(trade.type));
  
  // Group trades by candle interval (HH:MM)
  const tradesByInterval = new Map<string, typeof openTrades>();
  
  for (const trade of openTrades) {
    const interval = getCandleInterval(trade.timestamp);
    if (!tradesByInterval.has(interval)) {
      tradesByInterval.set(interval, []);
    }
    tradesByInterval.get(interval)!.push(trade);
  }

  // Check each interval for consecutive trades in the same direction
  const violations: Array<{ interval: string; trades: typeof openTrades }> = [];
  
  for (const [interval, intervalTrades] of tradesByInterval.entries()) {
    if (intervalTrades.length < 2) continue; // Need at least 2 trades to have consecutive
    
    // Check for consecutive trades in the same direction
    for (let i = 0; i < intervalTrades.length - 1; i++) {
      const currentTrade = intervalTrades[i];
      const nextTrade = intervalTrades[i + 1];
      
      const currentIsLong = isLongTrade(currentTrade.type);
      const nextIsLong = isLongTrade(nextTrade.type);
      
      // If both are the same direction (both long or both short), this is a violation
      if (currentIsLong === nextIsLong) {
        violations.push({
          interval,
          trades: [currentTrade, nextTrade],
        });
      }
    }
  }

  // Assert that no violations occurred
  // This test SHOULD FAIL on the current codebase (red state)

    ).join("; ")
  ).toHaveLength(0);

  // Also verify zero console errors
  expect(
    consoleErrors,
    `Expected no console errors, but found ${consoleErrors.length} error(s): ${consoleErrors.join("; ")}`
  ).toHaveLength(0);
});
