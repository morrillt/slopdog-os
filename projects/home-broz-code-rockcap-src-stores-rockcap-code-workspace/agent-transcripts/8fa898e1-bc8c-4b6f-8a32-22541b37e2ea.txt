## Basic Setup

1. **Install Tailscale** on both computers:
   - Linux: `sudo apt install tailscale` (or your distro's package manager)
   - Or download from https://tailscale.com/download

2. **Sign in** on both machines:
   ```bash
   sudo tailscale up
   ```
   This opens a browser to authenticate with your Tailscale account (Google, Microsoft, GitHub, etc.).

3. **Verify connection**:
   ```bash
   tailscale status
   ```
   Lists all devices in your Tailscale network with their IPs.

## Connecting to the Other Computer

Once both are connected:

- **SSH**: `ssh user@<tailscale-ip>` (use the Tailscale IP from `tailscale status`)
- **File sharing**: Access via the Tailscale IP (e.g., `smb://<tailscale-ip>`)
- **Other services**: Use the Tailscale IP instead of localhost/127.0.0.1

## Useful Commands

- `tailscale status` - See all connected devices
- `tailscale ip` - Get this machine's Tailscale IP
- `tailscale ping <hostname>` - Test connectivity
- `sudo tailscale down` - Disconnect
- `sudo tailscale up` - Reconnect

Both machines must be signed into the same Tailscale account and have Tailscale running. The connection is automatic once authenticated.

```bash:/home/broz/scripts/qwen-ollama
#!/bin/bash

# Wrapper script for using Qwen CLI with local Ollama models
# This script configures the Qwen CLI to connect to your local Ollama service via Tailscale

# Configuration: Set your Qwen server Tailscale hostname here (or leave empty for auto-discovery)
QWEN_SERVER_HOSTNAME="${QWEN_SERVER_HOSTNAME:-}"
FALLBACK_IP="192.168.1.70"
OLLAMA_PORT="11434"

# Function to test connectivity to Ollama server
test_ollama_connection() {
    local host="$1"
    local timeout=2
    curl -s --max-time "$timeout" "http://${host}:${OLLAMA_PORT}/api/tags" > /dev/null 2>&1
}

# Function to discover Qwen server via Tailscale
discover_qwen_server() {
    # If hostname is explicitly set, use it
    if [ -n "$QWEN_SERVER_HOSTNAME" ]; then
        echo "$QWEN_SERVER_HOSTNAME"
        return 0
    fi
    
    # Try to find server via Tailscale status
    if command -v tailscale > /dev/null 2>&1; then
        # Look for hostnames containing "qwen" or "ollama" (case-insensitive)
        local hostname=$(tailscale status 2>/dev/null | grep -iE "(qwen|ollama)" | head -1 | awk '{print $1}')
        if [ -n "$hostname" ]; then
            echo "$hostname"
            return 0
        fi
        
        # If no specific hostname found, try to get Tailscale IPs and test them
        local tailscale_ips=$(tailscale status 2>/dev/null | grep -v "^$" | grep -v "^100\." | awk '{print $1}' | grep -E "^[0-9]+\.[0-9]+\.[0-9]+\.[0-9]+$")
        for ip in $tailscale_ips; do
            if test_ollama_connection "$ip"; then
                echo "$ip"
                return 0
            fi
        done
    fi
    
    return 1
}

# Function to find working Ollama host
find_ollama_host() {
    local host=""
    
    # Try Tailscale discovery first
    if host=$(discover_qwen_server); then
        if test_ollama_connection "$host"; then
            echo "$host"
            return 0
        fi
    fi
    
    # Fall back to hardcoded IP
    if test_ollama_connection "$FALLBACK_IP"; then
        echo "$FALLBACK_IP"
        return 0
    fi
    
    return 1
}

# Find the Ollama host
OLLAMA_HOST_IP=$(find_ollama_host)

if [ -z "$OLLAMA_HOST_IP" ]; then
    echo "Error: Could not connect to Ollama server" >&2
    echo "Tried:" >&2
    echo "  - Tailscale discovery" >&2
    echo "  - Fallback IP: $FALLBACK_IP" >&2
    echo "" >&2
    echo "Please ensure:" >&2
    echo "  1. Tailscale is running and connected" >&2
    echo "  2. Your Qwen server is online and running Ollama" >&2
    echo "  3. Set QWEN_SERVER_HOSTNAME environment variable if needed" >&2
    exit 1
fi

OLLAMA_HOST="http://${OLLAMA_HOST_IP}:${OLLAMA_PORT}/v1"

# Display usage if no arguments provided
if [ $# -eq 0 ]; then
    echo "Usage: $0 [qwen-options]"
    echo "Example: $0 -p \"Hello, how are you?\" --model llama3"
    echo ""
    echo "Connected to Ollama server at: $OLLAMA_HOST_IP"
    echo ""
    echo "Available Ollama models:"
    curl -s "http://${OLLAMA_HOST_IP}:${OLLAMA_PORT}/api/tags" | jq -r '.models[]?.name' 2>/dev/null || echo "Could not fetch model list - is your Ollama server reachable?"
    exit 1
fi

# Run Qwen CLI with Ollama configuration
OPENAI_BASE_URL="$OLLAMA_HOST" OPENAI_API_KEY="ollama" qwen "$@"
```
