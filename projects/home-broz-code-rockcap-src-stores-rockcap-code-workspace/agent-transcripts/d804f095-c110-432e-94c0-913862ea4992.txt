 *
 * Common wait functions used across e2e tests for page loading and data availability.
 *
 * @module tests/e2e/helper/wait-helpers
 */

import { Page } from "@playwright/test";

/**
 * Waits for the backtest page to finish loading basic data.
 * Verifies that the chart container is visible and summary start price is loaded.
 *
 * @param page - Playwright page object
 */
export async function waitForStart(page: Page) {
  await page.waitForSelector('[data-testid="chart-container"]');
  await expect(page.getByTestId("summary-start")).not.toHaveText("–");
}

/**
 * Waits for the backtest page to finish loading and algo entry to be calculated.
 * Extends waitForStart to ensure algo-derived entry price is available.
 *
 * @param page - Playwright page object
 */
export async function waitForAlgoEntry(page: Page) {
  await waitForStart(page);
  // Wait for algo entry to be calculated (not "No trade")
  await expect(page.getByTestId("entry-input")).not.toHaveText("No trade", { timeout: 10000 });
}

/**
 * Waits for backtest page to be fully loaded with chart data.
 * Alternative to waitForStart with slightly different timing expectations.
 *
 * @param page - Playwright page object
 */
export async function waitForBacktestLoaded(page: Page) {
  await page.getByTestId("chart-container").waitFor({ state: "visible" });
  await expect(page.getByTestId("summary-start")).not.toHaveText("–");
}

/**
 * Waits for algorithmic trades to be generated and displayed.
 * Ensures trade rows are present in the algo trades table.
 *
 * @param page - Playwright page object
 */
export async function waitForAlgoTrades(page: Page) {
  // Wait for chart to load and algo trades to generate
  await page.waitForSelector('[data-testid="chart-container"]');
  await expect(page.getByTestId("summary-start")).not.toHaveText("–");
  // Wait for trade rows to appear (should be 2 rows)
  await page.waitForSelector('[data-testid="trade-row"]', { timeout: 10000 });
}

import { Page } from "@playwright/test";

/**
 * Gets the current backtest state from the browser's global store.
 * Useful for validating internal state in e2e tests.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to the backtest state object
 */
export async function getBacktestState(page: Page) {
  return await page.evaluate(() => {
    return window.__backtestState;
  });
}

/**
 * Gets the current algorithmic trade rows from the store.
 * Returns the generatedAlgoRows array which contains the trade data.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to array of AlgoTradeRow objects
 */
export async function getAlgoTradeRows(page: Page) {
  const state = await getBacktestState(page);
  return state?.generatedAlgoRows || [];
}

/**
 * Gets the current strategy results from localStorage.
 * Returns the stored strategy results data for multi-day analysis.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to strategy results object or null
 */
export async function getStrategyResults(page: Page) {
  return await page.evaluate(() => {
    try {
      const stored = localStorage.getItem("rockcap:strategyResults");
      return stored ? JSON.parse(stored) : null;
    } catch (e) {
      return null;
    }
  });
}

/**
 * Gets the current backtest parameters from the store.
 * Useful for validating that presets and parameters are set correctly.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to current backtest parameters
 */
export async function getBacktestParams(page: Page) {
  const state = await getBacktestState(page);
  return {
    profitTargetPts: state?.profitTargetPts,
    stopLossPts: state?.stopLossPts,
    triggerPts: state?.triggerPts,
    strategy: state?.strategy,
    selectedStrategyId: state?.selectedStrategyId,
    maxRoundTrips: state?.maxRoundTrips,
    cutoffTime: state?.cutoffTime,
    stopLossByProfitThreshold: state?.stopLossByProfitThreshold,
  };
}

/**
 * Gets the current chart series type (candles or trades).
 * Useful for validating chart display state.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to current series type
 */
export async function getChartSeries(page: Page) {
  const state = await getBacktestState(page);
  return state?.series || "candles";
}

/**
 * Gets the current table tab selection (algo, ohlcv, or trades).
 * Useful for validating which table is currently displayed.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to current table tab
 */
export async function getTableTab(page: Page) {
  const state = await getBacktestState(page);
  return state?.tableTab || "algo";
}

/**
 * Gets the current entry price and direction from the store.
 * Returns the algo-derived or user-set entry information.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to entry price and direction
 */
export async function getEntryInfo(page: Page) {
  const state = await getBacktestState(page);
  return {
    entryPrice: state?.entryPrice,
    direction: state?.direction,
  };
}

/**
 * Gets the total P&L from algorithmic trades.
 * Calculates the sum of all trade nets for validation.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to total P&L
 */
export async function getAlgoTotalPnL(page: Page) {
  const trades = await getAlgoTradeRows(page);

  // Sum up net values from close rows
  let total = 0;
  for (const trade of trades) {
    if (trade.type.includes('Close') && trade.net !== undefined) {
      total += trade.net;
    }
  }

  return total;
}

export async function getAlgoTotalPnL(page: Page) {
  const trades = await getAlgoTradeRows(page);

  // Sum up net values from close rows
  let total = 0;
  for (const trade of trades) {
    if (trade.type.includes('Close') && trade.net !== undefined) {
      total += trade.net;
    }
  }

  return total;
}

/**
 * Gets the count of round-trip trades from algo trade rows.
 * Each complete trade consists of one open and one close row.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to number of round-trip trades
 */
export async function getRoundTripCount(page: Page) {
  const trades = await getAlgoTradeRows(page);

  // Count close rows (each represents one completed round-trip)
  return trades.filter(trade => trade.type.includes('Close')).length;
}

/**
 * Gets the latest trade entry price from the algo trades table.
 * Returns the price from the most recent Open row.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to latest entry price or null
 */
export async function getLatestEntryPrice(page: Page) {
  const trades = await getAlgoTradeRows(page);

  // Find the most recent Open row
  const openRows = trades.filter(trade => trade.type.includes('Open'));
  const latestOpen = openRows[openRows.length - 1];

  return latestOpen?.price || null;
}

/**
 * Gets the exit reason for the latest completed trade.
 * Returns the note from the most recent Close row.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to exit reason or null
 */
export async function getLatestExitReason(page: Page) {
  const trades = await getAlgoTradeRows(page);

  // Find the most recent Close row
  const closeRows = trades.filter(trade => trade.type.includes('Close'));
  const latestClose = closeRows[closeRows.length - 1];

  return latestClose?.note || null;
}

/**
 * Validates that algo trade rows form proper open/close pairs.
 * Each open should have a corresponding close with proper timing.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to validation result
 */
export async function validateAlgoTradePairs(page: Page) {
  const trades = await getAlgoTradeRows(page);

  if (trades.length % 2 !== 0) {
    return { valid: false, reason: `Odd number of trades: ${trades.length}` };
  }

  // Check that trades alternate Open/Close
  for (let i = 0; i < trades.length; i++) {
    const expectedType = i % 2 === 0 ? 'Open' : 'Close';
    if (!trades[i].type.includes(expectedType)) {
      return { valid: false, reason: `Trade ${i} should be ${expectedType}, got ${trades[i].type}` };
    }
  }

  // Check that timestamps are monotonic
  for (let i = 1; i < trades.length; i++) {
    const prevTime = new Date(trades[i - 1].timestamp).getTime();
    const currTime = new Date(trades[i].timestamp).getTime();
    if (currTime < prevTime) {
      return { valid: false, reason: `Non-monotonic timestamps at trade ${i}` };
    }
  }

  return { valid: true };
}

/**
 * Gets strategy results summary from localStorage.
 * Returns aggregated stats for multi-day performance tracking.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to strategy results summary
 */
export async function getStrategyResultsSummary(page: Page) {
  const results = await getStrategyResults(page);

  if (!results) {
    return { days: 0, totalPnL: 0, avgPnL: 0 };
  }

  const days = Object.keys(results).length;
  const totalPnL = Object.values(results).reduce((sum: number, day: any) =>
    sum + (day.pnl || 0), 0);
  const avgPnL = days > 0 ? totalPnL / days : 0;

  return { days, totalPnL, avgPnL };
}

/**
 * Validates that algo trade rows form proper open/close pairs.
 * Each open should have a corresponding close with proper timing.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to validation result
 */
async function validateAlgoTradePairs(page) {
  const trades = await getAlgoTradeRows(page);

  if (trades.length % 2 !== 0) {
    return { valid: false, reason: `Odd number of trades: ${trades.length}` };
  }

  // Check that trades alternate Open/Close
  for (let i = 0; i < trades.length; i++) {
    const expectedType = i % 2 === 0 ? 'Open' : 'Close';
    if (!trades[i].type.includes(expectedType)) {
      return { valid: false, reason: `Trade ${i} should be ${expectedType}, got ${trades[i].type}` };
    }
  }

  // Check that timestamps are monotonic
  for (let i = 1; i < trades.length; i++) {
    const prevTime = new Date(trades[i - 1].timestamp).getTime();
    const currTime = new Date(trades[i].timestamp).getTime();
    if (currTime < prevTime) {
      return { valid: false, reason: `Non-monotonic timestamps at trade ${i}` };
    }
  }

  return { valid: true };
}

/**
 * Gets strategy results summary from localStorage.
 * Returns aggregated stats for multi-day performance tracking.
 *
 * @param page - Playwright page object
 * @returns Promise resolving to strategy results summary
 */
async function getStrategyResultsSummary(page) {
  const results = await getStrategyResults(page);

  if (!results) {
    return { days: 0, totalPnL: 0, avgPnL: 0 };
  }

  const days = Object.keys(results).length;
  const totalPnL = Object.values(results).reduce((sum, day) =>
    sum + (day.pnl || 0), 0);
  const avgPnL = days > 0 ? totalPnL / days : 0;

  return { days, totalPnL, avgPnL };
}

  };

async function useStrategyPerformance(page) {
  const summary = await getStrategyResultsSummary(page);

    ...summary,

  };

    useStrategyPerformance(page),
  ]);

  return {

    entry,
    trades,

    isReady: config.hasValidParams && entry.hasEntry,
    hasCompleteWorkflow: config.hasValidParams && entry.hasEntry && trades.hasTrades,
