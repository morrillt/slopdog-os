import path from "node:path";
import yaml from "js-yaml";
import { NextRequest, NextResponse } from "next/server";

const REPO_ROOT = path.resolve(process.cwd(), "..");

function parseFrontMatter(content: string): { frontmatter: any; body: string } {
  const frontMatterRegex = /^---\s*\n([\s\S]*?)\n---\s*\n([\s\S]*)$/;
  const match = content.match(frontMatterRegex);
  
  if (!match) {
    return { frontmatter: null, body: content };
  }

  try {
    const frontmatter = yaml.load(match[1]) as any;
    return { frontmatter, body: match[2] };
  } catch (err) {
    return { frontmatter: null, body: content };
  }
}

export async function POST(request: NextRequest) {
  try {
    const body = await request.json();
    const { filePath, updates } = body;

    if (!filePath || !updates) {
      return NextResponse.json(
        { error: "filePath and updates are required" },
        { status: 400 }
      );
    }

    // Security: ensure path is within docs/ or plans/
    const normalizedPath = path.normalize(filePath);
    if (!normalizedPath.startsWith("docs/") && !normalizedPath.startsWith("plans/")) {
      return NextResponse.json({ error: "Invalid path" }, { status: 400 });
    }

    // Prevent directory traversal
    if (normalizedPath.includes("..")) {
      return NextResponse.json({ error: "Invalid path" }, { status: 400 });
    }

    const fullPath = path.join(REPO_ROOT, normalizedPath);

    if (!fs.existsSync(fullPath)) {
      return NextResponse.json({ error: "File not found" }, { status: 404 });
    }

    // Read current content
    const currentContent = fs.readFileSync(fullPath, "utf-8");
    const { frontmatter, body } = parseFrontMatter(currentContent);

    // Create updated frontmatter
    const updatedFrontmatter = frontmatter || {};
    
    // Update tags if provided
    if (updates.tags !== undefined) {
      updatedFrontmatter.tags = updates.tags;
    }

    // Update facets if provided
    if (updates.facets !== undefined) {
      updatedFrontmatter.facets = {
        ...updatedFrontmatter.facets,
        ...updates.facets,
      };
    }

    // Update other fields if provided
    if (updates.title !== undefined) {
      updatedFrontmatter.title = updates.title;
    }
    if (updates.description !== undefined) {
      updatedFrontmatter.description = updates.description;
    }
    if (updates.updated !== undefined) {
      updatedFrontmatter.updated = updates.updated;
    }

    // Convert frontmatter back to YAML
    const frontMatterYaml = yaml.dump(updatedFrontmatter, {
      lineWidth: -1,
      noRefs: true,
      sortKeys: false,
    });

    // Reconstruct file content
    const updatedContent = `---\n${frontMatterYaml}---\n${body}`;

    // Write back to file
    fs.writeFileSync(fullPath, updatedContent, "utf-8");

    return NextResponse.json({
      success: true,
      path: normalizedPath,
    });
  } catch (err: any) {
    console.error("Error updating doc:", err);
    return NextResponse.json(
      { error: err?.message ?? "Failed to update document" },
      { status: 500 }
    );
  }
}
