  const searchInput = page.locator('input[placeholder="Search by title..."]');
  if (await searchInput.count() === 0) {

  await page.waitForTimeout(500);
  
  const suggestions = page.locator('.absolute.z-10 button');
  const count = await suggestions.count();
  const texts = [];
  for (let i = 0; i < Math.min(count, 5); i++) {
    texts.push(await suggestions.nth(i).textContent());

  return {
    searchValue: await searchInput.inputValue(),
    suggestionCount: count,
    suggestions: texts
  };
})(page);
