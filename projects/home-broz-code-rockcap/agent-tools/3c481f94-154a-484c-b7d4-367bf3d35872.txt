+function parseCsvLine(line) {
+  const fields = [];
+  let currentField = "";
+  let inQuotedField = false;
+  let i = 0;
+
+  while (i < line.length) {
+    const char = line[i];
+
+    if (char === '"') {
+      if (inQuotedField && i + 1 < line.length && line[i + 1] === '"') {
+        currentField += '"';
+        i += 2;
+      } else {
+        inQuotedField = !inQuotedField;
+        i++;
+      }
+    } else if (char === "," && !inQuotedField) {
+      fields.push(currentField);
+      currentField = "";
+      i++;
+    } else {
+      currentField += char;
+      i++;
+    }
+  }
+
+  fields.push(currentField);
+  return fields;
+}

+function selectTopConfigsPerStrategy(rows, topN, strategiesAllowlist) {

+function calculateMaxDrawdown(dailyPnL) {
+  let maxDrawdown = 0;
+  let peakDate = null;
+  let troughDate = null;
+  let recoveryDate = null;
+  let currentPeak = 0;
+  let currentPeakDate = null;
+  let inDrawdown = false;
+
+  for (const day of dailyPnL) {
+    const cumulative = day.cumulative;
+    if (cumulative > currentPeak) {
+      if (inDrawdown && recoveryDate === null) recoveryDate = day.date;
+      currentPeak = cumulative;
+      currentPeakDate = day.date;
+      inDrawdown = false;
+    } else {
+      inDrawdown = true;
+      const drawdown = currentPeak - cumulative;
+      if (drawdown > maxDrawdown) {
+        maxDrawdown = drawdown;
+        peakDate = currentPeakDate;
+        troughDate = day.date;
+        recoveryDate = null;
+      }
+    }
+  }
+
+  return { maxDrawdown, peakDate, troughDate, recoveryDate };
+}

+function calculateWinRate(trades) {
+  const closedTrades = trades.filter((t) => t.result !== "EOD Push");
+  if (closedTrades.length === 0) return 0;
+  const wins = closedTrades.filter((t) => t.result === "PT Hit").length;
+  return (wins / closedTrades.length) * 100;
+}

+function calculateProfitFactor(trades) {
+  const closedTrades = trades.filter((t) => t.result !== "EOD Push");
+  let grossProfit = 0;
+  let grossLoss = 0;
+
+  for (const trade of closedTrades) {
+    if (trade.net > 0) grossProfit += trade.net;
+    else grossLoss += Math.abs(trade.net);
+  }
+
+  if (grossLoss === 0) return grossProfit > 0 ? Infinity : 0;
+  return grossProfit / grossLoss;
+}

+function calculateConsecutiveLossStreak(trades) {
+  const closedTrades = trades.filter((t) => t.result !== "EOD Push");
+  let maxStreak = 0;
+  let currentStreak = 0;
+
+  for (const trade of closedTrades) {
+    if (trade.net < 0) {
+      currentStreak++;
+      maxStreak = Math.max(maxStreak, currentStreak);
+    } else {
+      currentStreak = 0;
+    }
+  }
+
+  return maxStreak;
+}

+function calculateSharpeRatio(dailyPnL) {
+  if (dailyPnL.length < 2) return 0;
+  const returns = dailyPnL.map((d) => d.netProfit);
+  const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
+  const variance = returns.reduce((sum, r) => sum + Math.pow(r - mean, 2), 0) / returns.length;
+  const stdDev = Math.sqrt(variance);
+  if (stdDev === 0) return 0;
+  return (mean / stdDev) * Math.sqrt(252);
+}

+function calculateDrawdownProbabilities(dailyPnL, thresholds = [50, 75, 100]) {
+  const probabilities = {};
+
+  for (const threshold of thresholds) {
+    let occurrences = 0;
+    let currentPeak = 0;
+    let drawdownStarted = false;
+
+    for (const day of dailyPnL) {
+      const cumulative = day.cumulative;
+      if (cumulative > currentPeak) {
+        currentPeak = cumulative;
+        drawdownStarted = false;
+      } else {
+        const drawdown = currentPeak - cumulative;
+        if (drawdown >= threshold && !drawdownStarted) {
+          occurrences++;
+          drawdownStarted = true;
+        }
+      }
+    }
+
+    probabilities[threshold] = (occurrences / dailyPnL.length) * 100;
+  }
+
+  return probabilities;
+}
